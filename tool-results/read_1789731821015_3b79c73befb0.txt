    57→    57→export function fmtMoney(n: number, code = "USD", fx: Fx = { USD: 1 }): string {
    58→    58→  const rate = code === "USD" ? 1 : fx[code] || 1;
    59→    59→  const value = n * rate;
    60→    60→  const abs = Math.abs(value);
    61→    61→  // small converted figures keep more decimals so prices never collapse
    62→    62→  const decimals = abs > 0 && abs < 1 ? 4 : abs < 100 ? 2 : 2;
    63→    63→  const body = value.toLocaleString("en-US", { minimumFractionDigits: decimals, maximumFractionDigits: decimals });
    64→    64→  const meta = CURRENCY_META[code];
    65→    65→  if (meta?.symbol && code === "USD") return `$${body}`;
    66→    66→  if (meta?.symbol) return `${meta.symbol}${body}`;
    67→    67→  return `${code} ${body}`;
    68→    68→}

    79→    79→export function fmtDate(iso: string, lang: Lang = "en"): string {
    80→    80→  const d = new Date(`${iso}T00:00:00`);
    81→    81→  if (Number.isNaN(d.getTime())) return iso;
    82→    82→  try {
    83→    83→    return d.toLocaleDateString(lang === "ar" ? "ar" : "en-US", { month: "short", day: "numeric", year: "numeric" });
    84→    84→  } catch {
    85→    85→    return iso;
    86→    86→  }
    87→    87→}

    89→    89→export function seedFor(id: string): number {
    90→    90→  let s = 0;
    91→    91→  for (let i = 0; i < id.length; i++) s = (s * 31 + id.charCodeAt(i)) % 233280;
    92→    92→  return s || 7;
    93→    93→}

   224→   224→export function PerformanceChart({ history, t, money = usd }: { history: number[]; t: T; money?: (n: number) => string }) {
   225→   225→  const pts = history.length > 1 ? history : [1, 1];
   226→   226→  const max = Math.max(...pts);
   227→   227→  const min = Math.min(...pts);
   228→   228→  const range = max - min || Math.max(1, max * 0.002);
   229→   229→  const W = 560;
   230→   230→  const H = 190;
   231→   231→  const step = W / (pts.length - 1);
   232→   232→  const coords = pts.map((p, i) => [i * step, H - 26 - ((p - min) / range) * (H - 52)] as const);
   233→   233→  let path = `M${coords[0][0]},${coords[0][1]}`;
   234→   234→  for (let i = 1; i < coords.length; i++) {
   235→   235→    const [x0, y0] = coords[i - 1];
   236→   236→    const [x1, y1] = coords[i];
   237→   237→    const cx = (x0 + x1) / 2;
   238→   238→    path += ` C${cx},${y0} ${cx},${y1} ${x1},${y1}`;
   239→   239→  }
   240→   240→  const area = `${path} L${W},${H} L0,${H} Z`;
   241→   241→  const last = coords[coords.length - 1];
   242→   242→
   243→   243→  return (
   244→   244→    <Card className="overflow-hidden">
   245→   245→      <SectionTitle title={t("performance")} right={<LiveChip t={t} />} />
   246→   246→      <div className="px-2 pb-3 pt-4">
   247→   247→        <svg viewBox={`0 0 ${W} ${H}`} className="h-36 w-full sm:h-44" preserveAspectRatio="none" role="img" aria-label="Portfolio performance chart">
   248→   248→          <defs>
   249→   249→            <linearGradient id="perfFill" x1="0" y1="0" x2="0" y2="1">
   250→   250→              <stop offset="0%" stopColor="#059669" stopOpacity="0.22" />
   251→   251→              <stop offset="100%" stopColor="#059669" stopOpacity="0.01" />
   252→   252→            </linearGradient>
   253→   253→          </defs>
   254→   254→          {[0.25, 0.5, 0.75].map((f) => (
   255→   255→            <line key={f} x1="0" x2={W} y1={26 + f * (H - 52)} y2={26 + f * (H - 52)} stroke="rgba(15,23,42,0.07)" strokeWidth="1" />
   256→   256→          ))}
   257→   257→          <path d={area} fill="url(#perfFill)" />
   258→   258→          <path d={path} fill="none" stroke="#059669" strokeWidth="2.2" strokeLinecap="round" />
   259→   259→          <circle cx={last[0]} cy={last[1]} r="3.5" fill="#059669" stroke="#ffffff" strokeWidth="1.5" />
   260→   260→          <text x={8} y={16} fontSize="11" fill="rgba(100,116,139,0.9)" className="font-mono" direction="ltr">
   261→   261→            {money(max)}
   262→   262→          </text>
   263→   263→          <text x={8} y={H - 6} fontSize="11" fill="rgba(100,116,139,0.9)" className="font-mono" direction="ltr">
   264→   264→            {money(min)}
   265→   265→          </text>
   266→   266→        </svg>
   267→   267→      </div>
   268→   268→    </Card>
   269→   269→  );
   270→   270→}

   280→   280→
   281→   281→export function buildHoldingRows(holdings: ClientView["client"]["holdings"], markets: Coin[]): HoldingRow[] {
   282→   282→  return holdings
   283→   283→    .map((h) => {
   284→   284→      const coin = markets.find((m) => m.id === h.assetId);
   285→   285→      if (!coin) return null;
   286→   286→      const base = sparkline(seedFor(coin.id), 24, 1.15);
   287→   287→      const trend = coin.change24h >= 0 ? 7 : -7;
   288→   288→      const series = base.map((v, i) => v + (i / (base.length - 1)) * trend);
   289→   289→      return { units: h.units, coin, value: coin.price * h.units, sparkPath: sparklinePath(series, 72, 28) };
   290→   290→    })
   291→   291→    .filter((r): r is HoldingRow => r !== null)
   292→   292→    .sort((a, b) => b.value - a.value);
   293→   293→}
