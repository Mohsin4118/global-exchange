    31→function downloadCsv(filename: string, rows: string[][]) {
    32→  const blob = new Blob([rows.map((r) => r.map((c) => `"${String(c).replace(/"/g, '""')}"`).join(",")).join("\n")], { type: "text/csv;charset=utf-8" });
    33→  const url = URL.createObjectURL(blob);
    34→  const a = document.createElement("a");
    35→  a.href = url;
    36→  a.download = filename;
    37→  a.click();
    38→  URL.revokeObjectURL(url);
    39→}

   294→function NewTxModal({ open, onClose, ctx }: { open: boolean; onClose: () => void; ctx: AdminCtx }) {
   295→  const [clientId, setClientId] = useState("");
   296→  const [type, setType] = useState<TxType>("CREDIT");
   297→  const [kind, setKind] = useState<TxKind>("deposit");
   298→  const [amount, setAmount] = useState("");
   299→  const [status, setStatus] = useState<TxStatus>("COMPLETED");
   300→  const [date, setDate] = useState(todayISO());
   301→  const [method, setMethod] = useState("Bank Transfer");
   302→  const [notes, setNotes] = useState("");
   303→
   304→  const submit = async () => {
   305→    const value = parseFloat(amount);
   306→    if (!clientId) {
   307→      ctx.toast("Select a client", "Choose which client this transaction belongs to.");
   308→      return;
   309→    }
   310→    if (!value || value <= 0) {
   311→      ctx.toast("Invalid amount", "Enter an amount greater than zero.");
   312→      return;
   313→    }
   314→    const ok = await ctx.runAction(
   315→      { action: "create-transaction", tx: { clientId, type, kind, amount: value, status, dateISO: date, method, notes } },
   316→      { title: "Transaction created", description: `${type === "CREDIT" ? "Credit" : "Debit"} of $${value.toLocaleString("en-US")} recorded — all dashboards updated.` },
   317→    );
   318→    if (ok) {
   319→      setClientId(""); setType("CREDIT"); setKind("deposit"); setAmount(""); setStatus("COMPLETED"); setDate(todayISO()); setMethod("Bank Transfer"); setNotes("");
   320→      onClose();
   321→    }
   322→  };
   323→
   324→  return (
   325→    <Modal open={open} onClose={onClose} title="New Transaction">
   326→      <div className="space-y-4">
   327→        <div>
   328→          <span className="mb-1.5 block text-[13px] font-medium text-slate-600">Client *</span>
   329→          <Select
   330→            value={clientId}
   331→            onChange={setClientId}
   332→            placeholder="Select client"
   333→            options={ctx.state.clients.map((c) => ({ value: c.id, label: c.name }))}
   334→          />
   335→        </div>
   336→        <div className="grid grid-cols-2 gap-4">
   337→          <div>
   338→            <span className="mb-1.5 block text-[13px] font-medium text-slate-600">Type *</span>
   339→            <Select
   340→              value={type}
   341→              onChange={(v) => {
   342→                setType(v as TxType);
   343→                setKind(v === "CREDIT" ? "deposit" : "withdrawal");
   344→              }}
   345→              options={[
   346→                { value: "CREDIT", label: "Credit (money in)" },
   347→                { value: "DEBIT", label: "Debit (money out)" },
   348→              ]}
   349→            />
   350→          </div>
   351→          <div>
   352→            <span className="mb-1.5 block text-[13px] font-medium text-slate-600">Kind</span>
   353→            <Select value={kind} onChange={(v) => setKind(v as TxKind)} options={KIND_OPTIONS} />
   354→          </div>
   355→          <TextInput label="Amount (USD) *" type="number" value={amount} onChange={setAmount} placeholder="0.00" />
   356→          <div>
   357→            <span className="mb-1.5 block text-[13px] font-medium text-slate-600">Status</span>
   358→            <Select value={status} onChange={(v) => setStatus(v as TxStatus)} options={TX_STATUS_OPTIONS} />
   359→          </div>
   360→          <TextInput label="Date" type="date" value={date} onChange={setDate} />
   361→          <TextInput label="Method" value={method} onChange={setMethod} />
   362→        </div>
   363→        <TextInput label="Notes" value={notes} onChange={setNotes} placeholder="Optional" />
   364→        <div className="flex justify-end gap-3 pt-2">
   365→          <OutlineButton onClick={onClose}>Cancel</OutlineButton>
   366→          <PrimaryButton onClick={submit}>Create Transaction</PrimaryButton>
   367→        </div>
   368→      </div>
   369→    </Modal>
   370→  );
   371→}

   373→function EditTxModal({ tx, onClose, ctx }: { tx: TxRow | null; onClose: () => void; ctx: AdminCtx }) {
   374→  const [dateISO, setDateISO] = useState("");
   375→  const [kind, setKind] = useState<TxKind>("deposit");
   376→  const [type, setType] = useState<TxType>("CREDIT");
   377→  const [amount, setAmount] = useState("");
   378→  const [status, setStatus] = useState<TxStatus>("COMPLETED");
   379→  const [label, setLabel] = useState("");
   380→  const [method, setMethod] = useState("");
   381→  const [notes, setNotes] = useState("");
   382→  const [loadedId, setLoadedId] = useState<string | null>(null);
   383→
   384→  if (tx && loadedId !== tx.id) {
   385→    setLoadedId(tx.id);
   386→    setDateISO(tx.dateISO);
   387→    setKind(tx.kind);
   388→    setType(tx.type);
   389→    setAmount(String(tx.amount));
   390→    setStatus(tx.status);
   391→    setLabel(tx.label);
   392→    setMethod(tx.method);
   393→    setNotes(tx.notes === "—" ? "" : tx.notes);
   394→  }
   395→
   396→  const close = () => {
   397→    setLoadedId(null);
   398→    onClose();
   399→  };
   400→
   401→  const save = async () => {
   402→    if (!tx) return;
   403→    const value = parseFloat(amount);
   404→    if (!Number.isFinite(value) || value <= 0) {
   405→      ctx.toast("Invalid amount", "Enter an amount greater than zero.");
   406→      return;
   407→    }
   408→    const ok = await ctx.runAction(
   409→      { action: "update-transaction", id: tx.id, patch: { dateISO, kind, type, amount: value, status, label, method, notes } },
   410→      { title: "Transaction updated", description: "Every balance, statistic and statement now reflects this change." },
   411→    );
   412→    if (ok) close();
   413→  };
   414→
   415→  return (
   416→    <Modal open={!!tx} onClose={close} title="Edit Transaction" panelClassName="max-w-lg">
   417→      {tx && (
   418→        <div className="space-y-4">
   419→          <p className="rounded-lg bg-slate-50 px-3.5 py-2.5 text-[12.5px] text-slate-500">
   420→            {tx.clientName} · <span className="font-mono">{tx.reference}</span>
   421→          </p>
   422→          <div className="grid grid-cols-2 gap-4">
   423→            <TextInput label="Date" type="date" value={dateISO} onChange={setDateISO} />
   424→            <TextInput label="Amount (USD) *" type="number" value={amount} onChange={setAmount} />
   425→            <div>
   426→              <span className="mb-1.5 block text-[13px] font-medium text-slate-600">Direction</span>
   427→              <Select value={type} onChange={(v) => setType(v as TxType)} options={[{ value: "CREDIT", label: "Credit (money in)" }, { value: "DEBIT", label: "Debit (money out)" }]} />
   428→            </div>
   429→            <div>
   430→              <span className="mb-1.5 block text-[13px] font-medium text-slate-600">Kind</span>
   431→              <Select value={kind} onChange={(v) => setKind(v as TxKind)} options={KIND_OPTIONS} />
   432→            </div>
   433→            <div>
   434→              <span className="mb-1.5 block text-[13px] font-medium text-slate-600">Status</span>
   435→              <Select value={status} onChange={(v) => setStatus(v as TxStatus)} options={TX_STATUS_OPTIONS} />
   436→            </div>
   437→            <TextInput label="Method" value={method} onChange={setMethod} />
   438→          </div>
   439→          <TextInput label="Description" value={label} onChange={setLabel} />
   440→          <TextInput label="Notes" value={notes} onChange={setNotes} placeholder="—" />
