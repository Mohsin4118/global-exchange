    34→export function usd(n: number): string {

    42→export function fmtDate(iso: string, lang: Lang = "en"): string {
    43→  const d = new Date(`${iso}T00:00:00`);
    44→  if (Number.isNaN(d.getTime())) return iso;
    45→  try {
    46→    return d.toLocaleDateString(lang === "ar" ? "ar" : "en-US", { month: "short", day: "numeric", year: "numeric" });
    47→  } catch {
    48→    return iso;
    49→  }
    50→}

    52→export function seedFor(id: string): number {
    53→  let s = 0;
    54→  for (let i = 0; i < id.length; i++) s = (s * 31 + id.charCodeAt(i)) % 233280;
    55→  return s || 7;
    56→}

    58→export function initials(name: string): string {
    59→  const parts = name.trim().split(/\s+/).slice(0, 2);
    60→  return parts.map((p) => p[0]?.toUpperCase() ?? "").join("") || "C";
    61→}

   155→export function PerformanceChart({ history, t }: { history: number[]; t: T }) {
   156→  const pts = history.length > 1 ? history : [1, 1];
   157→  const max = Math.max(...pts);
   158→  const min = Math.min(...pts);
   159→  const range = max - min || Math.max(1, max * 0.002);
   160→  const W = 560;
   161→  const H = 190;
   162→  const step = W / (pts.length - 1);
   163→  const coords = pts.map((p, i) => [i * step, H - 26 - ((p - min) / range) * (H - 52)] as const);
   164→  let path = `M${coords[0][0]},${coords[0][1]}`;
   165→  for (let i = 1; i < coords.length; i++) {
   166→    const [x0, y0] = coords[i - 1];
   167→    const [x1, y1] = coords[i];
   168→    const cx = (x0 + x1) / 2;
   169→    path += ` C${cx},${y0} ${cx},${y1} ${x1},${y1}`;
   170→  }
   171→  const area = `${path} L${W},${H} L0,${H} Z`;
   172→  const last = coords[coords.length - 1];
   173→
   174→  return (
   175→    <Card className="overflow-hidden">
   176→      <SectionTitle title={t("performance")} right={<LiveChip t={t} />} />
   177→      <div className="px-2 pb-3 pt-4">
   178→        <svg viewBox={`0 0 ${W} ${H}`} className="h-40 w-full sm:h-44" preserveAspectRatio="none" role="img" aria-label="Portfolio performance chart">
   179→          <defs>
   180→            <linearGradient id="perfFill" x1="0" y1="0" x2="0" y2="1">
   181→              <stop offset="0%" stopColor="#059669" stopOpacity="0.22" />
   182→              <stop offset="100%" stopColor="#059669" stopOpacity="0.01" />
   183→            </linearGradient>
   184→          </defs>
   185→          {[0.25, 0.5, 0.75].map((f) => (
   186→            <line key={f} x1="0" x2={W} y1={26 + f * (H - 52)} y2={26 + f * (H - 52)} stroke="rgba(15,23,42,0.07)" strokeWidth="1" />
   187→          ))}
   188→          <path d={area} fill="url(#perfFill)" />
   189→          <path d={path} fill="none" stroke="#059669" strokeWidth="2.2" strokeLinecap="round" />
   190→          <circle cx={last[0]} cy={last[1]} r="3.5" fill="#059669" stroke="#ffffff" strokeWidth="1.5" />
   191→          <text x={8} y={16} fontSize="11" fill="rgba(100,116,139,0.9)" className="font-mono" direction="ltr">
   192→            {usd(max)}
   193→          </text>
   194→          <text x={8} y={H - 6} fontSize="11" fill="rgba(100,116,139,0.9)" className="font-mono" direction="ltr">
   195→            {usd(min)}
   196→          </text>
   197→        </svg>
   198→      </div>
   199→    </Card>
   200→  );
   201→}

   205→export interface HoldingRow {
   206→  units: number;
   207→  coin: Coin;
   208→  value: number;
   209→  sparkPath: string;
   210→}
   211→
   212→export function buildHoldingRows(holdings: ClientView["client"]["holdings"], markets: Coin[]): HoldingRow[] {
   213→  return holdings
   214→    .map((h) => {
   215→      const coin = markets.find((m) => m.id === h.assetId);
   216→      if (!coin) return null;
   217→      const base = sparkline(seedFor(coin.id), 24, 1.15);
   218→      const trend = coin.change24h >= 0 ? 7 : -7;
   219→      const series = base.map((v, i) => v + (i / (base.length - 1)) * trend);
   220→      return { units: h.units, coin, value: coin.price * h.units, sparkPath: sparklinePath(series, 72, 28) };
   221→    })
   222→    .filter((r): r is HoldingRow => r !== null)
   223→    .sort((a, b) => b.value - a.value);
   224→}
