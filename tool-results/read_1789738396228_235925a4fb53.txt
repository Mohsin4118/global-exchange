    39→export function hashPassword(password: string): { salt: string; hash: string } {
    40→  const salt = crypto.randomBytes(16).toString("hex");
    41→  const hash = crypto.scryptSync(password, salt, 64).toString("hex");
    42→  return { salt, hash };
    43→}
    44→
    45→export function verifyPassword(password: string, stored: { salt: string; hash: string }): boolean {
    46→  try {
    47→    const candidate = crypto.scryptSync(password, stored.salt, 64);
    48→    const expected = Buffer.from(stored.hash, "hex");
    49→    return candidate.length === expected.length && crypto.timingSafeEqual(candidate, expected);
    50→  } catch {
    51→    return false;
    52→  }
    53→}

    57→let seq = 0;
    58→export function uid(prefix: string): string {
    59→  seq += 1;
    60→  return `${prefix}${Date.now().toString(36)}${seq.toString(36)}${crypto.randomBytes(3).toString("hex")}`;
    61→}

    63→export function newReference(): string {
    64→  const d = new Date();
    65→  const ymd = `${d.getFullYear().toString().slice(-2)}${String(d.getMonth() + 1).padStart(2, "0")}${String(d.getDate()).padStart(2, "0")}`;
    66→  return `TXN-${ymd}-${crypto.randomInt(1000, 9999)}`;
    67→}

    69→export function stamp(): string {
    70→  const d = new Date();
    71→  const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
    72→  let h = d.getHours();
    73→  const ampm = h >= 12 ? "PM" : "AM";
    74→  h = h % 12 || 12;
    75→  return `${months[d.getMonth()]} ${d.getDate()}, ${d.getFullYear()}, ${String(h).padStart(2, "0")}:${String(d.getMinutes()).padStart(2, "0")} ${ampm}`;
    76→}

    78→function relTime(iso: string): string {
    79→  const then = new Date(iso).getTime();
    80→  if (Number.isNaN(then)) return "just now";
    81→  const mins = Math.max(0, Math.round((Date.now() - then) / 60000));
    82→  if (mins < 1) return "just now";
    83→  if (mins < 60) return `${mins}m ago`;
    84→  const hours = Math.round(mins / 60);
    85→  if (hours < 24) return `${hours}h ago`;
    86→  const days = Math.round(hours / 24);
    87→  if (days < 7) return `${days}d ago`;
    88→  return `${Math.round(days / 7)}w ago`;
    89→}
    90→
    91→function daysAgoISO(days: number): string {
    92→  const d = new Date();
    93→  d.setDate(d.getDate() - days);
    94→  return d.toISOString().slice(0, 10);
    95→}

    97→function hashCode(s: string): number {
    98→  let h = 0;
    99→  for (let i = 0; i < s.length; i++) h = (h * 31 + s.charCodeAt(i)) | 0;
   100→  return h;
   101→}

   105→function makeTx(p: {
   106→  clientId: string;
   107→  dateISO: string;

   109→  type: Tx["type"];
   110→  amount: number;

   120→    id: uid("tx"),
   121→    clientId: p.clientId,
   122→    dateISO: p.dateISO,
   123→    kind: p.kind,
   124→    type: p.type,
   125→    amount: p.amount,
   126→    label: p.label,
   127→    labelKey: p.labelKey,
   128→    asset: p.asset,
   129→    status,
   130→    reference: newReference(),
   131→    method: p.method ?? "Bank Transfer",
   132→    notes: p.notes ?? "—",
   133→    history: [{ at: new Date(`${p.dateISO}T10:00:00Z`).toISOString(), by: "CryptoWise", byRole: "system", from: null, to: status }],
   134→    createdAtISO: new Date(`${p.dateISO}T10:00:00Z`).toISOString(),
   135→    updatedAtISO: new Date(`${p.dateISO}T10:00:00Z`).toISOString(),
   136→  };
   137→}

   147→    createdAtISO,

   151→}

   429→function readDb(): DbData {
   430→  if (!fs.existsSync(DB_FILE)) {
   431→    fs.mkdirSync(DATA_DIR, { recursive: true });
   432→    const seeded = seed();
   433→    fs.writeFileSync(DB_FILE, JSON.stringify(seeded, null, 2), "utf8");
   434→    return seeded;
   435→  }
   436→  try {
   437→    const raw = fs.readFileSync(DB_FILE, "utf8");
   438→    const parsed = JSON.parse(raw) as DbData;
   439→    if (!parsed.adminUser || !Array.isArray(parsed.clients)) throw new Error("corrupt db");
   440→    parsed.sessions ??= [];
   441→    parsed.comments ??= [];
   442→    parsed.staff ??= [];
   443→    parsed.roles ??= [];
   444→    parsed.audit ??= [];
   445→    parsed.notifications ??= [];
   446→    parsed.txs ??= [];
   447→    // display-currency backfill (older db files keep working)
   448→    parsed.fx = { ...DEFAULT_FX, ...(parsed.fx ?? {}) };
   449→    for (const c of parsed.clients) {
   450→      c.currency ??= "USD";
   451→      c.sourceOfFunds ??= "";
   452→    }
   453→    return parsed;
   454→  } catch {
   455→    const seeded = seed();
   456→    fs.writeFileSync(DB_FILE, JSON.stringify(seeded, null, 2), "utf8");
   457→    return seeded;
   458→  }
   459→}

   461→function writeDb(data: DbData) {
   462→  fs.mkdirSync(DATA_DIR, { recursive: true });
   463→  const tmp = `${DB_FILE}.tmp`;
   464→  fs.writeFileSync(tmp, JSON.stringify(data, null, 2), "utf8");
   465→  fs.renameSync(tmp, DB_FILE);
   466→}

   468→/** Read (seeding on first use). */
   469→export function db(): DbData {
   470→  return readDb();
   471→}

   473→/** Mutate the database atomically and persist. */
   474→export function mutate<T>(fn: (data: DbData) => T): T {
   475→  const data = readDb();
   476→  const result = fn(data);
   477→  writeDb(data);
   478→  return result;
   479→}

   525→  return Math.round((n + Number.EPSILON) * 100) / 100;

   528→/** Request statuses that are still moving through the pipeline. Everything
   529→    except REJECTED / CANCELLED / COMPLETED reserves funds (withdrawals) and
   530→    counts as an open request. */
   531→export function inFlight(status: TxStatus): boolean {
   532→  return status === "PENDING" || status === "UNDER_REVIEW" || status === "APPROVED" || status === "PROCESSING";
   533→}

   535→export function computeFinancials(clientId: string, data: DbData): ComputedFinancials {
   536→  const client = data.clients.find((c) => c.id === clientId);
   537→  const own = data.txs.filter((t) => t.clientId === clientId);
   538→  let credits = 0;
   539→  let debits = 0;
   540→  let pendingWithdrawals = 0;
   541→  for (const t of own) {
   542→    if (t.status === "COMPLETED") {
   543→      if (t.type === "CREDIT") credits += t.amount;
   544→      else debits += t.amount;
   545→    } else if (inFlight(t.status) && t.kind === "withdrawal" && t.type === "DEBIT") {
   546→      pendingWithdrawals += t.amount;
   547→    }
   548→  }
   549→  const balance = Math.max(0, (client?.openingBalance ?? 0) + credits - debits);
   550→  return {
   551→    balance: round2(balance),
   552→    available: round2(Math.max(0, balance - pendingWithdrawals)),
   553→    credits: round2(credits),
   554→    debits: round2(debits),
   555→    pendingWithdrawals: round2(pendingWithdrawals),
   556→    txCount: own.length,
   557→  };
   558→}

   560→/** Balance timeline per client so each tx row can show "balance after". */
   561→function balanceAfterMap(data: DbData): Map<string, number> {
   562→  const result = new Map<string, number>();
   563→  const byClient = new Map<string, Tx[]>();
   564→  for (const t of data.txs) {
   565→    const list = byClient.get(t.clientId) ?? [];
   566→    list.push(t);
   567→    byClient.set(t.clientId, list);
   568→  }
   569→  for (const [clientId, list] of byClient) {
   570→    const client = data.clients.find((c) => c.id === clientId);
   571→    const sorted = [...list].sort((a, b) => (a.dateISO === b.dateISO ? a.createdAtISO.localeCompare(b.createdAtISO) : a.dateISO.localeCompare(b.dateISO)));
   572→    let bal = client?.openingBalance ?? 0;
   573→    for (const t of sorted) {
   574→      if (t.status === "COMPLETED") bal = Math.max(0, bal + (t.type === "CREDIT" ? t.amount : -t.amount));
   575→      result.set(t.id, round2(bal));
   576→    }
   577→  }
   578→  return result;
   579→}

   588→export function adminSnapshot(): AdminSnapshot {
   589→  const data = readDb();
   590→  const balAfter = balanceAfterMap(data);
   591→  const clients = data.clients.map((c) => ({
   592→    ...publicClient(c),
   593→    financials: computeFinancials(c.id, data),
   594→    hasPortalPassword: true,
   595→  }));
   596→  const nameById = new Map(data.clients.map((c) => [c.id, c.name] as const));
   597→  const txs = [...data.txs]
   598→    .sort((a, b) => (a.dateISO === b.dateISO ? b.createdAtISO.localeCompare(a.createdAtISO) : b.dateISO.localeCompare(a.dateISO)))
   599→    .map((t) => ({ ...t, clientName: nameById.get(t.clientId) ?? "—", balanceAfter: balAfter.get(t.id) ?? 0 }));
   600→
   601→  const completed = txs.filter((t) => t.status === "COMPLETED");
   602→  const requestCounts = txs.reduce(
   603→    (acc, t) => {
   604→      acc[t.status] = (acc[t.status] ?? 0) + 1;
   605→      return acc;
   606→    },
   607→    { COMPLETED: 0, PENDING: 0, UNDER_REVIEW: 0, APPROVED: 0, PROCESSING: 0, REJECTED: 0, CANCELLED: 0 } as Record<TxStatus, number>,
   608→  );
   609→  const stats: AdminStats = {
   610→    totalClients: clients.length,
   611→    activeClients: clients.filter((c) => c.status === "active").length,
   612→    totalBalance: round2(clients.reduce((s, c) => s + c.financials.balance, 0)),
   613→    volume: round2(completed.reduce((s, t) => s + t.amount, 0)),
   614→    credits: round2(completed.filter((t) => t.type === "CREDIT").reduce((s, t) => s + t.amount, 0)),
   615→    debits: round2(completed.filter((t) => t.type === "DEBIT").reduce((s, t) => s + t.amount, 0)),
   616→    pendingWithdrawals: txs.filter((t) => t.kind === "withdrawal" && inFlight(t.status)).length,
   617→    txCount: txs.length,
   618→    requestCounts,
   619→  };
   620→
   621→  const comments: Record<string, ClientComment[]> = {};
   622→  for (const cm of data.comments) {
   623→    (comments[cm.clientId] ??= []).push(cm);
   624→  }
   625→
   626→  return {
   627→    clients,
   628→    txs,
   629→    notifications: [...data.notifications].sort((a, b) => b.createdAtISO.localeCompare(a.createdAtISO)),
   630→    audit: data.audit,
   631→    comments,
   632→    staff: data.staff,
   633→    roles: data.roles,
   634→    stats,
   635→    fx: data.fx,
   636→  };
   637→}

   640→export function clientView(clientId: string): { ok: true; view: { client: PublicClient; financials: ComputedFinancials; txs: Tx[]; notifications: Notification[]; fx: Record<string, number> } } | { ok: false } {

   654→      fx: data.fx,
   655→    },
   656→  };
   657→}

   659→/** Client-safe transaction: strips the PRIVATE admin note and every internal
   660→    remark from the history (the status transition itself stays visible; only
   661→    the internal wording is removed). Admin actors are anonymised to
   662→    "Super Admin" so admin emails/details never leak through the client API (#30 / #36). */
   663→export function clientSafeTx(t: Tx): Tx {
   664→  const { adminNote: _private, ...rest } = t;
   665→  const history = (t.history ?? []).map<TxEvent>((e) => ({
   666→    at: e.at,
   667→    by: e.byRole === "admin" ? "Super Admin" : e.by,
   668→    byRole: e.byRole,
   669→    from: e.from,
   670→    to: e.to,
   671→    ...(e.note ? { note: e.note } : {}),
   672→    ...(e.changes ? { changes: e.changes } : {}),
   673→  }));
   674→  return { ...rest, history };
   675→}
