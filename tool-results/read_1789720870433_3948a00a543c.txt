    34→export function usd(n: number): string {

    42→export function fmtDate(iso: string, lang: Lang = "en"): string {
    43→  const d = new Date(`${iso}T00:00:00`);
    44→  if (Number.isNaN(d.getTime())) return iso;
    45→  try {
    46→    return d.toLocaleDateString(lang === "ar" ? "ar" : "en-US", { month: "short", day: "numeric", year: "numeric" });
    47→  } catch {
    48→    return iso;
    49→  }
    50→}

    52→export function seedFor(id: string): number {
    53→  let s = 0;
    54→  for (let i = 0; i < id.length; i++) s = (s * 31 + id.charCodeAt(i)) % 233280;
    55→  return s || 7;
    56→}

    58→export function initials(name: string): string {
    59→  const parts = name.trim().split(/\s+/).slice(0, 2);
    60→  return parts.map((p) => p[0]?.toUpperCase() ?? "").join("") || "C";
    61→}

   144→export function PerformanceChart({ history, t }: { history: number[]; t: T }) {
   145→  const pts = history.length > 1 ? history : [1, 1];
   146→  const max = Math.max(...pts);
   147→  const min = Math.min(...pts);
   148→  const range = max - min || Math.max(1, max * 0.002);
   149→  const W = 560;
   150→  const H = 190;
   151→  const step = W / (pts.length - 1);
   152→  const coords = pts.map((p, i) => [i * step, H - 26 - ((p - min) / range) * (H - 52)] as const);
   153→  let path = `M${coords[0][0]},${coords[0][1]}`;
   154→  for (let i = 1; i < coords.length; i++) {
   155→    const [x0, y0] = coords[i - 1];
   156→    const [x1, y1] = coords[i];
   157→    const cx = (x0 + x1) / 2;
   158→    path += ` C${cx},${y0} ${cx},${y1} ${x1},${y1}`;
   159→  }
   160→  const area = `${path} L${W},${H} L0,${H} Z`;
   161→  const last = coords[coords.length - 1];
   162→
   163→  return (
   164→    <Card className="overflow-hidden">
   165→      <SectionTitle title={t("performance")} right={<LiveChip t={t} />} />
   166→      <div className="px-2 pb-3 pt-4">
   167→        <svg viewBox={`0 0 ${W} ${H}`} className="h-40 w-full sm:h-44" preserveAspectRatio="none" role="img" aria-label="Portfolio performance chart">
   168→          <defs>
   169→            <linearGradient id="perfFill" x1="0" y1="0" x2="0" y2="1">
   170→              <stop offset="0%" stopColor="#059669" stopOpacity="0.22" />
   171→              <stop offset="100%" stopColor="#059669" stopOpacity="0.01" />
   172→            </linearGradient>
   173→          </defs>
   174→          {[0.25, 0.5, 0.75].map((f) => (
   175→            <line key={f} x1="0" x2={W} y1={26 + f * (H - 52)} y2={26 + f * (H - 52)} stroke="rgba(15,23,42,0.07)" strokeWidth="1" />
   176→          ))}
   177→          <path d={area} fill="url(#perfFill)" />
   178→          <path d={path} fill="none" stroke="#059669" strokeWidth="2.2" strokeLinecap="round" />
   179→          <circle cx={last[0]} cy={last[1]} r="3.5" fill="#059669" stroke="#ffffff" strokeWidth="1.5" />
   180→          <text x={8} y={16} fontSize="11" fill="rgba(100,116,139,0.9)" className="font-mono" direction="ltr">
   181→            {usd(max)}
   182→          </text>
   183→          <text x={8} y={H - 6} fontSize="11" fill="rgba(100,116,139,0.9)" className="font-mono" direction="ltr">
   184→            {usd(min)}
   185→          </text>
   186→        </svg>
   187→      </div>
   188→    </Card>
   189→  );
   190→}

   194→export interface HoldingRow {
   195→  units: number;
   196→  coin: Coin;
   197→  value: number;
   198→  sparkPath: string;

   200→
   201→export function buildHoldingRows(holdings: ClientView["client"]["holdings"], markets: Coin[]): HoldingRow[] {
   202→  return holdings
   203→    .map((h) => {
   204→      const coin = markets.find((m) => m.id === h.assetId);
   205→      if (!coin) return null;
   206→      const base = sparkline(seedFor(coin.id), 24, 1.15);
   207→      const trend = coin.change24h >= 0 ? 7 : -7;
   208→      const series = base.map((v, i) => v + (i / (base.length - 1)) * trend);
   209→      return { units: h.units, coin, value: coin.price * h.units, sparkPath: sparklinePath(series, 72, 28) };
   210→    })
   211→    .filter((r): r is HoldingRow => r !== null)
   212→    .sort((a, b) => b.value - a.value);
   213→}

   382→/* ---------------- request modal (deposit / withdrawal, dark) ---------------- */
   383→
   384→export function RequestModal({
   385→  kind,
   386→  t,
   387→  available,
   388→  onClose,
   389→  onSubmit,
   390→}: {
   391→  kind: "deposit" | "withdrawal";
   392→  t: T;
   393→  available: number;
   394→  onClose: () => void;
   395→  onSubmit: (amount: number, note: string) => Promise<string | null>; // returns error key or null
   396→}) {
   397→  const [amount, setAmount] = useState("");
   398→  const [note, setNote] = useState("");
   399→  const [error, setError] = useState<string | null>(null);
   400→  const [busy, setBusy] = useState(false);
   401→
   402→  const submit = async () => {
   403→    const value = parseFloat(amount);
   404→    if (!Number.isFinite(value) || value <= 0) {
   405→      setError(t("requestInvalid"));
   406→      return;
   407→    }
   408→    if (kind === "withdrawal" && value > available) {
   409→      setError(t("requestTooMuch"));
   410→      return;
   411→    }
   412→    setBusy(true);
   413→    const err = await onSubmit(value, note.trim());
   414→    setBusy(false);
   415→    if (err) setError(t(err as StringKey));
   416→  };
   417→
   418→  return (
   419→    <div className="fixed inset-0 z-50 flex items-center justify-center p-4">
   420→      <div className="absolute inset-0 bg-slate-900/40 backdrop-blur-[2px]" onClick={onClose} />
   421→      <div className="relative z-10 w-full max-w-md overflow-hidden rounded-2xl border border-slate-200 bg-white shadow-2xl">
   422→        <div className="flex items-center justify-between p-5 pb-0">
   423→          <h3 className="text-lg font-bold text-slate-900">{kind === "deposit" ? t("requestTitleDeposit") : t("requestTitleWithdraw")}</h3>
   424→          <button onClick={onClose} aria-label="Close" className="flex h-8 w-8 items-center justify-center rounded-lg text-slate-500 transition-colors hover:bg-slate-100 hover:text-slate-900">
   425→            <X className="h-5 w-5" />
   426→          </button>
   427→        </div>
   428→        <div className="space-y-4 p-5">
   429→          <p className="rounded-lg bg-white px-3.5 py-2.5 text-[12.5px] leading-relaxed text-slate-500">{t("requestHint")}</p>
   430→          <label className="block">
   431→            <span className="mb-1.5 block text-[13px] font-medium text-slate-600">{t("requestAmount")}</span>
   432→            <input
   433→              type="number"
   434→              min="0"
   435→              step="any"
   436→              value={amount}
   437→              onChange={(e) => setAmount(e.target.value)}
   438→              placeholder="0.00"
   439→              dir="ltr"
   440→              aria-label={t("requestAmount")}

   521→/* ---------------- profile & account ---------------- */
   522→
   523→export function ProfileCard({
   524→  view,
   525→  t,
   526→  onProfileSave,
   527→  onPasswordSave,
   528→  lang,
   529→}: {
   530→  view: ClientView;
   531→  t: T;
   532→  onProfileSave: (patch: { phone: string; country: string; address: string; city: string; postcode: string }) => Promise<string | null>;
   533→  onPasswordSave: (current: string, next: string) => Promise<string | null>;
   534→  lang: Lang;
   535→}) {
   536→  const c = view.client;
   537→  const fin = view.financials;
   538→  const [editing, setEditing] = useState(false);
   539→  const [form, setForm] = useState({ phone: c.phone, country: c.country, address: c.address, city: c.city, postcode: c.postcode });
   540→  const [busy, setBusy] = useState(false);
   541→  const [pwOpen, setPwOpen] = useState(false);
   542→  const [pw, setPw] = useState({ current: "", next: "", confirm: "" });
   543→  const [pwError, setPwError] = useState<string | null>(null);
   544→
   545→  const save = async () => {
   546→    setBusy(true);
   547→    const err = await onProfileSave(form);
   548→    setBusy(false);
   549→    if (!err) setEditing(false);
   550→  };
   551→
   552→  const changePassword = async () => {
   553→    if (pw.next !== pw.confirm) {
   554→      setPwError(t("passwordsMismatch"));
   555→      return;
   556→    }
   557→    setBusy(true);
   558→    const err = await onPasswordSave(pw.current, pw.next);
   559→    setBusy(false);
   560→    if (err) {
   561→      setPwError(t(err as StringKey));
   562→      return;
   563→    }
   564→    setPwOpen(false);
   565→    setPw({ current: "", next: "", confirm: "" });
   566→    setPwError(null);
   567→  };
   568→
   569→  const rows: Array<[string, string, boolean?]> = [
   570→    [t("emailAddress"), c.email, true],
   571→    [t("phone"), c.phone || "—", true],
   572→    [t("address"), [c.address, c.city, c.postcode].filter(Boolean).join(", ") || "—"],
   573→    [t("country"), c.country || "—"],
   574→    [t("accountNo"), c.accountNo, true],
   575→    [t("openingBalance"), usd(c.openingBalance), true],
   576→    [t("creditsLabel"), usd(fin.credits), true],
   577→    [t("debitsLabel"), usd(fin.debits), true],
   578→    [t("statusLabel"), c.status === "active" ? t("active") : t("suspended")],
   579→    [t("tierLabel"), c.tier === "Private" ? t("tierPrivate") : c.tier === "Premium" ? t("tierPremium") : t("tierStandard")],
   580→  ];
   581→
   582→  const inputCls =
   583→    "h-11 w-full rounded-lg border border-slate-200 bg-white px-3.5 text-sm text-slate-900 outline-none transition-colors placeholder:text-slate-400 focus:border-emerald-500 focus:ring-2 focus:ring-emerald-500/15";
   584→
   585→  return (
   586→    <Card className="overflow-hidden">
   587→      <SectionTitle
   588→        title={t("profileTitle")}
   589→        right={
   590→          !editing ? (
   591→            <button
   592→              onClick={() => {
   593→                setForm({ phone: c.phone, country: c.country, address: c.address, city: c.city, postcode: c.postcode });
   594→                setEditing(true);
   595→              }}
   596→              className="inline-flex items-center gap-1.5 rounded-lg border border-slate-200 px-2.5 py-1.5 text-[12px] font-semibold text-slate-600 transition-colors hover:bg-slate-100 hover:text-slate-900"
   597→            >
   598→              <Pencil className="h-3.5 w-3.5" /> {t("editProfile")}
   599→            </button>
   600→          ) : undefined
   601→        }
   602→      />
   603→      <div className="px-4 pb-5 pt-3 sm:px-5">
   604→        {editing ? (
   605→          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   606→            <label className="block">
   607→              <span className="mb-1.5 block text-[13px] font-medium text-slate-600">{t("phone")}</span>
   608→              <input value={form.phone} onChange={(e) => setForm({ ...form, phone: e.target.value })} dir="ltr" className={inputCls} />
   609→            </label>
   610→            <label className="block">
   611→              <span className="mb-1.5 block text-[13px] font-medium text-slate-600">{t("country")}</span>
   612→              <input value={form.country} onChange={(e) => setForm({ ...form, country: e.target.value })} className={inputCls} />
   613→            </label>
   614→            <label className="block sm:col-span-2">
   615→              <span className="mb-1.5 block text-[13px] font-medium text-slate-600">{t("address")}</span>
   616→              <input value={form.address} onChange={(e) => setForm({ ...form, address: e.target.value })} className={inputCls} />
   617→            </label>
   618→            <label className="block">
   619→              <span className="mb-1.5 block text-[13px] font-medium text-slate-600">{t("city")}</span>
   620→              <input value={form.city} onChange={(e) => setForm({ ...form, city: e.target.value })} className={inputCls} />
   621→            </label>
   622→            <label className="block">
   623→              <span className="mb-1.5 block text-[13px] font-medium text-slate-600">{t("postcode")}</span>
   624→              <input value={form.postcode} onChange={(e) => setForm({ ...form, postcode: e.target.value })} className={inputCls} />
   625→            </label>
   626→            <div className="flex justify-end gap-2.5 sm:col-span-2">
   627→              <button onClick={() => setEditing(false)} className="rounded-lg border border-slate-200 px-4 py-2.5 text-sm font-semibold text-slate-600 transition-colors hover:bg-slate-100">
   628→                {t("cancel")}
   629→              </button>
   630→              <button onClick={save} disabled={busy} className="rounded-lg bg-[#00E5A0] px-4 py-2.5 text-sm font-bold text-[#022c20] transition-colors hover:bg-[#2cf0b5] disabled:opacity-50">
   631→                {t("saveProfile")}
   632→              </button>
   633→            </div>
   634→          </div>
   635→        ) : (
   636→          <dl className="grid grid-cols-1 gap-x-8 sm:grid-cols-2">
   637→            {rows.map(([k, v, ltr]) => (
   638→              <div key={k} className="flex items-center justify-between gap-4 border-b border-slate-100 py-2.5 last:border-0">
   639→                <dt className="shrink-0 text-[12.5px] font-medium text-slate-500">{k}</dt>
   640→                <dd className="truncate text-[13px] font-semibold text-slate-900" dir={ltr || /^[+$\d]/.test(v) ? "ltr" : undefined}>
   641→                  {v}
   642→                </dd>
   643→              </div>
   644→            ))}
   645→          </dl>
   646→        )}
   647→
   648→        <div className="mt-5 border-t border-slate-200/70 pt-4">
   649→          {!pwOpen ? (
   650→            <button onClick={() => setPwOpen(true)} className="text-[13px] font-bold text-emerald-600 transition-colors hover:text-emerald-700">
   651→              {t("changePassword")}
   652→            </button>
   653→          ) : (
   654→            <div className="max-w-sm space-y-3">
   655→              <label className="block">
   656→                <span className="mb-1.5 block text-[13px] font-medium text-slate-600">{t("currentPassword")}</span>
   657→                <PasswordInput theme="light" value={pw.current} onChange={(v) => setPw({ ...pw, current: v })} autoComplete="current-password" />
   658→              </label>
   659→              <label className="block">
   660→                <span className="mb-1.5 block text-[13px] font-medium text-slate-600">{t("newPassword")}</span>
   661→                <PasswordInput theme="light" value={pw.next} onChange={(v) => setPw({ ...pw, next: v })} autoComplete="new-password" />
   662→              </label>
   663→              <label className="block">
   664→                <span className="mb-1.5 block text-[13px] font-medium text-slate-600">{t("confirmNewPassword")}</span>
   665→                <PasswordInput theme="light" value={pw.confirm} onChange={(v) => setPw({ ...pw, confirm: v })} autoComplete="new-password" />
   666→              </label>
   667→              {pwError && <p className="rounded-lg bg-amber-50 px-3.5 py-2 text-[12.5px] font-semibold text-amber-800 ring-1 ring-amber-600/15">{pwError}</p>}
   668→              <div className="flex justify-end gap-2.5">
   669→                <button
   670→                  onClick={() => {
   671→                    setPwOpen(false);
   672→                    setPwError(null);
   673→                  }}
   674→                  className="rounded-lg border border-slate-200 px-4 py-2.5 text-sm font-semibold text-slate-600 transition-colors hover:bg-slate-100"
   675→                >
   676→                  {t("cancel")}
   677→                </button>
   678→                <button onClick={changePassword} disabled={busy} className="rounded-lg bg-[#00E5A0] px-4 py-2.5 text-sm font-bold text-[#022c20] transition-colors hover:bg-[#2cf0b5] disabled:opacity-50">
   679→                  {t("changePassword")}
   680→                </button>
   681→              </div>
   682→            </div>
   683→          )}
   684→        </div>
   685→      </div>
   686→      {lang === "ar" && <span className="hidden">{c.name}</span>}
   687→    </Card>
   688→  );
   689→}
