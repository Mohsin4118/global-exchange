    67→export function ClientDashboard({

    84→  const [ready, setReady] = useState(false);
    85→  const [section, setSection] = useState<Section>("overview");

    87→  const [txFilter, setTxFilter] = useState<TxFilter>("all");
    88→  const [reqModal, setReqModal] = useState<null | RequestKind>(null);
    89→  const [showAllTxs, setShowAllTxs] = useState(false);
    90→
    91→  /* ---- live self-refresh: re-read the authorized API snapshot ---- */
    92→  const load = useCallback(async () => {
    93→    const res = await apiClientGet();
    94→    if (!res.ok) {
    95→      if (res.error === "unauthorized" || res.error === "gone") {
    96→        toast({ title: t("accountClosedToast") });
    97→        onSignOut();

    99→      return;
   100→    }
   101→    setView({ client: res.client!, financials: res.financials!, txs: res.txs!, notifications: res.notifications!, fx: res.fx! });
   102→    setReady(true);
   103→  }, [onSignOut, t, toast]);
   104→
   105→  useEffect(() => {
   106→    const boot = () => {
   107→      void load();
   108→    };
   109→    boot();
   110→    const id = setInterval(boot, 4000);
   111→    window.addEventListener("focus", boot);
   112→    return () => {
   113→      clearInterval(id);
   114→      window.removeEventListener("focus", boot);
   115→    };

   117→
   118→  /* The dashboard ALWAYS opens at the very top: session restore / browser
   119→     scroll restoration must never land the client mid-page with the welcome

   122→  useEffect(() => {
   123→    window.scrollTo({ top: 0 });
   124→  }, []);
   125→
   126→  /* ---- live portfolio math (server balance + live market prices) ---- */
   127→  const markets = useMemo(() => [...coins, ...stocks], [coins, stocks]);
   128→  const holdingRows = useMemo(() => (view ? buildHoldingRows(view.client.holdings, markets) : []), [view, markets]);

   130→  const cash = view?.financials.balance ?? 0;
   131→  const available = view?.financials.available ?? 0;
   132→  const total = cash + invested;
   133→  const portfolioChange = invested > 0 ? holdingRows.reduce((s, r) => s + r.coin.change24h * r.value, 0) / invested : 0;
   134→  const totalDeposits = view?.financials.credits ?? 0;
   135→  const totalWithdrawn = useMemo(() => (view ? view.txs.filter((x) => x.kind === "withdrawal" && x.status === "COMPLETED").reduce((s, x) => s + x.amount, 0) : 0), [view]);
   136→
   137→  const suspended = view?.client.status === "suspended";
   138→  const unread = view?.notifications.filter((n) => n.unread).length ?? 0;
   139→
   140→  /* ---- display currency: one formatter drives every figure (#10) ---- */
   141→  const cur = view?.client.currency ?? "USD";
   142→  const fx = view?.fx ?? { USD: 1 };
   143→  const money = useMemo(() => makeMoney(cur, fx), [cur, fx]);

   145→    if (!CURRENCIES.includes(code)) return;
   146→    const res = await apiClientAction({ action: "set-currency", currency: code });
   147→    if (res.ok && res.client && res.txs) {
   148→      setView({ client: res.client, financials: res.financials!, txs: res.txs, notifications: res.notifications!, fx: res.fx! });
   149→    }
   150→  };
   151→
   152→  /* ---- session performance chart ---- */
   153→  const [history, setHistory] = useState<number[]>([]);
   154→  const lastTotalRef = useRef(total);
   155→  useEffect(() => {
   156→    setHistory((h) => {
   157→      if (h.length > 0 || total <= 0) return h;
   158→      const pts: number[] = [];
   159→      let v = total * 0.982;
   160→      let s = Math.max(1, Math.floor(total)) % 233280 || 13;
   161→      const rand = () => {
   162→        s = (s * 9301 + 49297) % 233280;
   163→        return s / 233280;
   164→      };
   165→      for (let i = 0; i < 42; i++) {
   166→        v += total * (rand() - 0.47) * 0.004;
   167→        pts.push(Math.max(total * 0.9, v));
   168→      }
   169→      pts[pts.length - 1] = total;
   170→      lastTotalRef.current = total;
   171→      return pts;
   172→    });
   173→  }, [total]);
   174→
   175→  useEffect(() => {
   176→    const append = () => {
   177→      if (history.length === 0) return;
   178→      if (Math.abs(total - lastTotalRef.current) < 0.005) return;
   179→      lastTotalRef.current = total;
   180→      setHistory((h) => [...h.slice(-79), total]);
   181→    };
   182→    append();
   183→  }, [total, history.length]);
   184→
   185→  /* ---- actions ---- */
   186→  const submitRequest = async (payload: { kind: RequestKind; amount: number; asset: string; destination: string; note: string }): Promise<string | null> => {
   187→    const res = await apiClientAction({ action: "request-transaction", ...payload });
   188→    if (!res.ok) {
   189→      return res.error === "funds" ? "requestTooMuch" : res.error === "duplicate" ? "requestDuplicate" : "requestInvalid";
   190→    }
   191→    if (res.client && res.txs) setView({ client: res.client, financials: res.financials!, txs: res.txs, notifications: res.notifications!, fx: res.fx! });
   192→    toast({ title: t("requestSent"), description: t("requestSentSub") });
   193→    setReqModal(null);
   194→    return null;
   195→  };
   196→
   197→  /* the ONLY mutation a client may make on a request: cancelling their own PENDING one */
   198→  const cancelRequest = async (tx: { id: string; reference: string }) => {

   200→    if (!res.ok) {
   201→      toast({ title: t("requestInvalid") });
   202→      return;
   203→    }
   204→    if (res.client && res.txs) setView({ client: res.client, financials: res.financials!, txs: res.txs, notifications: res.notifications!, fx: res.fx! });

   206→  };
   207→
   208→  const markAllRead = async () => {
   209→    const res = await apiClientAction({ action: "mark-read", ids: "all" });

   211→      setView({ client: res.client, financials: res.financials!, txs: res.txs, notifications: res.notifications!, fx: res.fx! });
   212→    }
   213→  };
   214→
   215→  const saveProfile = async (patch: { phone: string; country: string; address: string; city: string; postcode: string }): Promise<string | null> => {
   216→    const res = await apiClientAction({ action: "update-profile", ...patch });
   217→    if (!res.ok) return "requestInvalid";
   218→    if (res.client && res.txs) setView({ client: res.client, financials: res.financials!, txs: res.txs, notifications: res.notifications!, fx: res.fx! });
   219→    toast({ title: t("profileSaved") });
   220→    return null;
   221→  };
   222→
   223→  const savePassword = async (current: string, next: string): Promise<string | null> => {
   224→    const res = await apiClientAction({ action: "change-password", current, next });
   225→    if (!res.ok) return res.error === "wrong-current" ? "wrongCurrentPassword" : res.error === "weak" ? "weakPassword" : "requestInvalid";
   226→    toast({ title: t("passwordChangedToast") });
   227→    return null;
   228→  };
   229→
   230→  const signOut = async () => {
   231→    await apiLogout("client");
   232→    onSignOut();
   233→  };
   234→
   235→  const downloadCsv = () => {
   236→    if (!view) return;
   237→    const header = "Date,Description,Asset,Method,Type,Amount USD,Status,Reference\n";
   238→    const lines = view.txs.map((tx) =>
   239→      [tx.dateISO, `"${(tx.labelKey ? t(tx.labelKey as StringKey) : tx.label).replace(/"/g, '""')}"`, tx.asset ?? "", `"${tx.method}"`, tx.type, tx.amount.toFixed(2), tx.status, tx.reference].join(","),
   240→    );
   241→    const blob = new Blob([header + lines.join("\n")], { type: "text/csv;charset=utf-8" });
   242→    const url = URL.createObjectURL(blob);
   243→    const a = document.createElement("a");
   244→    a.href = url;
   245→    a.download = `cryptowise-statement-${view.client.accountNo}.csv`;
   246→    a.click();
   247→    URL.revokeObjectURL(url);
   248→    toast({ title: t("statementToast") });
   249→  };
   250→
   251→  /* ---- derived lists ---- */
   252→  const txs = view?.txs ?? [];
   253→  const filteredTxs = useMemo(() => txs.filter((tx) => txFilter === "all" || (txFilter === "in" ? tx.type === "CREDIT" : tx.type === "DEBIT")), [txs, txFilter]);
   254→  const deposits = useMemo(() => txs.filter((tx) => tx.kind === "deposit"), [txs]);
   255→  const withdrawals = useMemo(() => txs.filter((tx) => tx.kind === "withdrawal"), [txs]);
   256→  const pendingWithdrawalTotal = withdrawals.filter((w) => w.status === "PENDING" || w.status === "PROCESSING").reduce((s, w) => s + w.amount, 0);
   257→
   258→  const allocation = useMemo(() => {
   259→    const segs = holdingRows.map((r) => ({ label: r.coin.name, value: r.value, color: r.coin.color }));
   260→    if (cash > 0) segs.unshift({ label: t("cash"), value: cash, color: "#10b981" });
   261→    return segs;
   262→  }, [holdingRows, cash, t]);
   263→
   264→  /* ---- menu definition (every item works — switches sections) ---- */
   265→  const NAV: Array<{ id: Section; label: string; icon: React.ReactNode; badge?: number }> = [
   266→    { id: "overview", label: t("menuOverview"), icon: <LayoutDashboard className="h-4 w-4" /> },
   267→    { id: "portfolio", label: t("menuPortfolio"), icon: <PieChart className="h-4 w-4" /> },
   268→    { id: "transactions", label: t("menuTransactions"), icon: <ArrowLeftRight className="h-4 w-4" /> },
   269→    { id: "deposits", label: t("menuDeposits"), icon: <ArrowDownLeft className="h-4 w-4" /> },
   270→    { id: "withdrawals", label: t("menuWithdrawals"), icon: <ArrowUpRight className="h-4 w-4" /> },
   271→    { id: "notifications", label: t("menuNotifications"), icon: <Bell className="h-4 w-4" />, badge: unread },

   273→  ];

   275→  const go = (s: Section) => {

   277→    setDrawer(false);
   278→    window.scrollTo({ top: 0 });
   279→  };
   280→
   281→  if (!ready || !view) {
   282→    return (
   283→      <div className="flex min-h-screen items-center justify-center bg-[#f5f7fa]">
   284→        <div className="flex flex-col items-center gap-3">
   285→          <span className="h-9 w-9 animate-spin rounded-full border-[3px] border-emerald-200 border-t-emerald-600" />
   286→          <p className="text-sm font-semibold text-slate-400">CryptoWise</p>
   287→        </div>
   288→      </div>
   289→    );
   290→  }
   291→
   292→  const c = view.client;
   293→
   294→  return (
   295→    <div className="min-h-screen overflow-x-clip bg-[#f5f7fa] text-slate-900 [color-scheme:light]">
   296→
   297→      {/* ---------- topbar ---------- */}
   298→      <header className="sticky top-0 z-40 border-b border-slate-200/80 bg-white/90 backdrop-blur-xl">
   299→        <div className="mx-auto flex h-14 w-full max-w-[1280px] items-center justify-between gap-2 px-4 sm:h-16 sm:gap-3 sm:px-6">
   300→          <div className="flex min-w-0 items-center gap-2.5 sm:gap-3">
   301→            <button
   302→              onClick={() => setDrawer(true)}
   303→              aria-label="Open menu"
   304→              aria-expanded={drawer}
   305→              className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg border border-slate-200 text-slate-600 transition-colors hover:bg-slate-100 hover:text-slate-900 lg:hidden"
   306→            >
   307→              <Menu className="h-4.5 w-4.5" />
   308→            </button>
   309→            <LogoMark className="h-8 w-8 shrink-0 sm:h-9 sm:w-9" />
   310→            <div className="min-w-0">
   311→              <p className="truncate text-[13.5px] font-bold leading-tight text-slate-900 sm:text-[14px]">CryptoWise</p>
   312→              <p className="truncate text-[10.5px] font-semibold text-emerald-600 sm:text-[11px]">{t("clientArea")}</p>
   313→            </div>
   314→            <span className="ms-1 hidden shrink-0 rounded-full bg-slate-100 px-2.5 py-1 text-[11px] font-bold text-slate-500 ring-1 ring-white/10 md:inline-block" dir="ltr">
   315→              {t("accountNo")} {c.accountNo}
   316→            </span>
   317→          </div>
   318→          <div className="flex shrink-0 items-center gap-1.5 sm:gap-2">
   319→            <button
   320→              onClick={onLangToggle}
   321→              className="flex h-9 items-center rounded-lg border border-slate-200 px-2.5 text-[11.5px] font-bold text-slate-600 transition-colors hover:bg-slate-100 hover:text-slate-900 sm:px-3 sm:text-[12px]"
   322→              aria-label="Toggle language"
   323→            >
   324→              {lang === "en" ? "العربية" : "EN"}
   325→            </button>
   326→            <button
   327→              onClick={() => go("notifications")}
   328→              aria-label={t("notificationsTitle")}
   329→              className="relative flex h-9 w-9 items-center justify-center rounded-lg border border-slate-200 text-slate-600 transition-colors hover:bg-slate-100 hover:text-slate-900"
   330→            >
   331→              <Bell className="h-4 w-4" />
   332→              {unread > 0 && (
   333→                <span className="absolute -right-1 -top-1 flex h-4 min-w-4 items-center justify-center rounded-full bg-amber-400 px-1 text-[9.5px] font-bold text-[#022c20]">{unread}</span>
   334→              )}
   335→            </button>
   336→            <div className="hidden items-center gap-2.5 rounded-full border border-slate-200 py-1 pe-3 ps-1 sm:flex">
   337→              <span className="flex h-7 w-7 items-center justify-center rounded-full bg-[#00E5A0] text-[11px] font-bold text-[#022c20]">{initials(c.name)}</span>
   338→              <span className="hidden text-left leading-tight md:block" dir="auto">
   339→                <span className="block max-w-[120px] truncate text-[12px] font-bold text-slate-900">{c.name}</span>
   340→                <span className="block text-[10.5px] font-semibold text-slate-500">{t("privateClient")}</span>
   341→              </span>
   342→            </div>
   343→            <button
   344→              onClick={signOut}
   345→              aria-label={t("signOut")}
   346→              className="flex h-9 items-center gap-1.5 rounded-lg border border-slate-200 px-2.5 text-[12px] font-semibold text-slate-600 transition-colors hover:bg-slate-100 hover:text-slate-900 sm:px-3"
   347→            >
   348→              <LogOut className="h-3.5 w-3.5" />
   349→              <span className="hidden md:inline">{t("signOut")}</span>
   350→            </button>
   351→          </div>
   352→        </div>
   353→      </header>
   354→
   355→      {/* ---------- body: sidebar + content ---------- */}
   356→      <div className="relative mx-auto flex w-full max-w-[1280px] items-start gap-6 px-4 sm:px-6">
   357→        {/* desktop sidebar */}
   358→        <aside className="sticky top-[72px] hidden w-[218px] shrink-0 py-6 lg:block" aria-label="Client menu">
   359→          <nav className="overflow-hidden rounded-2xl border border-slate-200/80 bg-white shadow-sm p-2">
   360→            {NAV.map((item) => (
   361→              <button
   362→                key={item.id}
   363→                onClick={() => go(item.id)}
   364→                aria-current={section === item.id ? "page" : undefined}
   365→                className={cn(
   366→                  "flex w-full items-center gap-2.5 rounded-xl px-3 py-2.5 text-[13px] font-semibold transition-colors",
   367→                  section === item.id ? "bg-emerald-50 text-emerald-600" : "text-slate-600 hover:bg-slate-50 hover:text-slate-900",
   368→                )}
   369→              >
   370→                {item.icon}
   371→                <span className="flex-1 text-start">{item.label}</span>
   372→                {!!item.badge && item.badge > 0 && (
   373→                  <span className="flex h-4.5 min-w-[18px] items-center justify-center rounded-full bg-amber-400 px-1 text-[10px] font-bold text-[#022c20]">{item.badge}</span>
   374→                )}
   375→              </button>
   376→            ))}
   377→            <div className="my-2 h-px bg-slate-100" />
   378→            <button
   379→              onClick={signOut}
   380→              className="flex w-full items-center gap-2.5 rounded-xl px-3 py-2.5 text-[13px] font-semibold text-slate-600 transition-colors hover:bg-slate-50 hover:text-slate-900"
   381→            >
   382→              <LogOut className="h-4 w-4" />
   383→              <span className="flex-1 text-start">{t("signOut")}</span>
   384→            </button>
   385→          </nav>
   386→          <p className="mt-3 px-3 text-[10.5px] leading-relaxed text-slate-400" dir="ltr">
   387→            {t("accountNo")} {c.accountNo}
   388→          </p>
   389→        </aside>
   390→
   391→        {/* content */}
   392→        <main className="min-w-0 flex-1 py-4 sm:py-6">
   393→          {section === "overview" && (
   394→            <OverviewSection
   395→              view={view}
   396→              t={t}
   397→              lang={lang}
   398→              coins={coins}
   399→              stocks={stocks}
   400→              holdingRows={holdingRows}
   401→              invested={invested}
   402→              cash={cash}
   403→              total={total}
   404→              available={available}
   405→              portfolioChange={portfolioChange}
   406→              totalDeposits={totalDeposits}
   407→              history={history}
   408→              allocation={allocation}
   409→              suspended={!!suspended}
   410→              money={money}
   411→              cur={cur}
   412→              onCurrency={setCurrency}
   413→              onDeposit={() => !suspended && setReqModal("deposit")}
   414→              onWithdraw={() => !suspended && setReqModal("withdrawal")}
   415→              onGo={go}
   416→            />
   417→          )}
   418→
   419→          {section === "portfolio" && (
   420→            <div className="grid grid-cols-1 gap-5">
   421→              <SectionHeading title={t("menuPortfolio")} sub={t("welcomeSub")} />
   422→              <HoldingsTable rows={holdingRows} t={t} lang={lang} />
   423→              <AllocationBar segments={allocation} t={t} />
   424→              <MarketStrip coins={coins} stocks={stocks} t={t} />
   425→            </div>
   426→          )}
   427→
   428→          {section === "transactions" && (
   429→            <div className="grid grid-cols-1 gap-5">
   430→              <SectionHeading title={t("txHistory")} sub={`${txs.length} · ${t("allTime")}`} />
   431→              <div className="flex flex-wrap items-center justify-between gap-2.5">
   432→                <div className="flex items-center gap-1 rounded-lg bg-slate-100 p-1">
   433→                  {(
   434→                    [
   435→                      ["all", t("filterAll")],
   436→                      ["in", t("filterIn")],
   437→                      ["out", t("filterOut")],
   438→                    ] as const
   439→                  ).map(([key, label]) => (
   440→                    <button
   441→                      key={key}
   442→                      onClick={() => setTxFilter(key)}
   443→                      className={cn("rounded-md px-2.5 py-1.5 text-[11.5px] font-bold transition-colors", txFilter === key ? "bg-emerald-100 text-emerald-600" : "text-slate-500 hover:text-slate-900")}
   444→                    >
   445→                      {label}
   446→                    </button>
   447→                  ))}
   448→                </div>
   449→                <div className="flex items-center gap-2">
