    32→export function usd(n: number): string {

    40→export function fmtDate(iso: string, lang: Lang = "en"): string {
    41→  const d = new Date(`${iso}T00:00:00`);
    42→  if (Number.isNaN(d.getTime())) return iso;
    43→  try {
    44→    return d.toLocaleDateString(lang === "ar" ? "ar" : "en-US", { month: "short", day: "numeric", year: "numeric" });
    45→  } catch {
    46→    return iso;
    47→  }
    48→}
    49→
    50→export function seedFor(id: string): number {
    51→  let s = 0;
    52→  for (let i = 0; i < id.length; i++) s = (s * 31 + id.charCodeAt(i)) % 233280;
    53→  return s || 7;
    54→}
    55→
    56→export function initials(name: string): string {
    57→  const parts = name.trim().split(/\s+/).slice(0, 2);
    58→  return parts.map((p) => p[0]?.toUpperCase() ?? "").join("") || "C";
    59→}

   128→export function PerformanceChart({ history, t }: { history: number[]; t: T }) {
   129→  const pts = history.length > 1 ? history : [1, 1];
   130→  const max = Math.max(...pts);
   131→  const min = Math.min(...pts);
   132→  const range = max - min || Math.max(1, max * 0.002);
   133→  const W = 560;
   134→  const H = 190;
   135→  const step = W / (pts.length - 1);
   136→  const coords = pts.map((p, i) => [i * step, H - 26 - ((p - min) / range) * (H - 52)] as const);
   137→  let path = `M${coords[0][0]},${coords[0][1]}`;
   138→  for (let i = 1; i < coords.length; i++) {
   139→    const [x0, y0] = coords[i - 1];
   140→    const [x1, y1] = coords[i];
   141→    const cx = (x0 + x1) / 2;
   142→    path += ` C${cx},${y0} ${cx},${y1} ${x1},${y1}`;
   143→  }
   144→  const area = `${path} L${W},${H} L0,${H} Z`;
   145→  const last = coords[coords.length - 1];
   146→
   147→  return (
   148→    <Card className="overflow-hidden">
   149→      <SectionTitle
   150→        title={t("performance")}
   151→        right={
   152→          <span className="inline-flex items-center gap-1.5 rounded-full bg-emerald-50 px-2.5 py-1 text-[11px] font-bold text-emerald-700">
   153→            <span className="relative flex h-1.5 w-1.5">
   154→              <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-emerald-500 opacity-60" />
   155→              <span className="relative inline-flex h-1.5 w-1.5 rounded-full bg-emerald-500" />
   156→            </span>
   157→            {t("live")}
   158→          </span>
   159→        }
   160→      />
   161→      <div className="px-2 pb-3 pt-4">
   162→        <svg viewBox={`0 0 ${W} ${H}`} className="h-44 w-full" preserveAspectRatio="none" role="img" aria-label="Portfolio performance chart">
   163→          <defs>
   164→            <linearGradient id="perfFill" x1="0" y1="0" x2="0" y2="1">
   165→              <stop offset="0%" stopColor="#10b981" stopOpacity="0.18" />
   166→              <stop offset="100%" stopColor="#10b981" stopOpacity="0.01" />
   167→            </linearGradient>
   168→          </defs>
   169→          {[0.25, 0.5, 0.75].map((f) => (
   170→            <line key={f} x1="0" x2={W} y1={26 + f * (H - 52)} y2={26 + f * (H - 52)} stroke="#eef2f7" strokeWidth="1" />
   171→          ))}
   172→          <path d={area} fill="url(#perfFill)" />
   173→          <path d={path} fill="none" stroke="#10b981" strokeWidth="2.2" strokeLinecap="round" />
   174→          <circle cx={last[0]} cy={last[1]} r="3.5" fill="#10b981" stroke="#fff" strokeWidth="1.5" />
   175→          <text x={8} y={16} fontSize="11" className="fill-slate-400 font-mono" direction="ltr">
   176→            {usd(max)}
   177→          </text>
   178→          <text x={8} y={H - 6} fontSize="11" className="fill-slate-400 font-mono" direction="ltr">
   179→            {usd(min)}
   180→          </text>
   181→        </svg>
   182→      </div>
   183→    </Card>
   184→  );
   185→}

   189→export interface HoldingRow {
   190→  units: number;
   191→  coin: Coin;
   192→  value: number;
   193→  sparkPath: string;
   194→}
   195→
   196→export function buildHoldingRows(holdings: ClientView["client"]["holdings"], markets: Coin[]): HoldingRow[] {
   197→  return holdings
   198→    .map((h) => {
   199→      const coin = markets.find((m) => m.id === h.assetId);
   200→      if (!coin) return null;
   201→      const base = sparkline(seedFor(coin.id), 24, 1.15);
   202→      const trend = coin.change24h >= 0 ? 7 : -7;
   203→      const series = base.map((v, i) => v + (i / (base.length - 1)) * trend);
   204→      return { units: h.units, coin, value: coin.price * h.units, sparkPath: sparklinePath(series, 72, 28) };
   205→    })
   206→    .filter((r): r is HoldingRow => r !== null)
   207→    .sort((a, b) => b.value - a.value);
   208→}

   339→/* ---------------- request modal (deposit / withdrawal) ---------------- */
   340→
   341→export function RequestModal({
   342→  kind,
   343→  t,
   344→  available,
   345→  onClose,
   346→  onSubmit,
   347→}: {
   348→  kind: "deposit" | "withdrawal";
   349→  t: T;
   350→  available: number;
   351→  onClose: () => void;
   352→  onSubmit: (amount: number, note: string) => Promise<string | null>; // returns error key or null
   353→}) {
   354→  const [amount, setAmount] = useState("");
   355→  const [note, setNote] = useState("");
   356→  const [error, setError] = useState<string | null>(null);
   357→  const [busy, setBusy] = useState(false);
   358→
   359→  const submit = async () => {
   360→    const value = parseFloat(amount);
   361→    if (!Number.isFinite(value) || value <= 0) {
   362→      setError(t("requestInvalid"));
   363→      return;
   364→    }
   365→    if (kind === "withdrawal" && value > available) {
   366→      setError(t("requestTooMuch"));
   367→      return;
   368→    }
   369→    setBusy(true);
   370→    const err = await onSubmit(value, note.trim());
   371→    setBusy(false);
   372→    if (err) setError(t(err as StringKey));
   373→  };
   374→
   375→  return (
   376→    <div className="fixed inset-0 z-50 flex items-center justify-center p-4">
   377→      <div className="absolute inset-0 bg-slate-950/50 backdrop-blur-[2px]" onClick={onClose} />
   378→      <div className="relative z-10 w-full max-w-md overflow-hidden rounded-2xl bg-white shadow-2xl">
   379→        <div className="flex items-center justify-between p-5 pb-0">
   380→          <h3 className="text-lg font-bold text-slate-900">{kind === "deposit" ? t("requestTitleDeposit") : t("requestTitleWithdraw")}</h3>
   381→          <button onClick={onClose} aria-label="Close" className="flex h-8 w-8 items-center justify-center rounded-lg text-slate-400 transition-colors hover:bg-slate-100 hover:text-slate-700">
   382→            <X className="h-5 w-5" />
   383→          </button>
   384→        </div>
   385→        <div className="space-y-4 p-5">
   386→          <p className="rounded-lg bg-slate-50 px-3.5 py-2.5 text-[12.5px] leading-relaxed text-slate-500">{t("requestHint")}</p>
   387→          <label className="block">
   388→            <span className="mb-1.5 block text-[13px] font-medium text-slate-600">{t("requestAmount")}</span>
   389→            <input
   390→              type="number"
   391→              min="0"
   392→              step="any"
   393→              value={amount}
   394→              onChange={(e) => setAmount(e.target.value)}
   395→              placeholder="0.00"
   396→              dir="ltr"
   397→              aria-label={t("requestAmount")}
   398→              className="h-11 w-full rounded-lg border border-slate-200 px-3.5 text-sm text-slate-900 outline-none transition-colors placeholder:text-slate-400 focus:border-emerald-500 focus:ring-2 focus:ring-emerald-500/15"
   399→            />
   400→          </label>
   401→          <label className="block">
   402→            <span className="mb-1.5 block text-[13px] font-medium text-slate-600">{t("requestNote")}</span>
   403→            <textarea
   404→              rows={2}
   405→              value={note}
   406→              onChange={(e) => setNote(e.target.value)}
   407→              className="w-full resize-y rounded-lg border border-slate-200 px-3.5 py-2.5 text-sm text-slate-900 outline-none transition-colors placeholder:text-slate-400 focus:border-emerald-500 focus:ring-2 focus:ring-emerald-500/15"
   408→            />
   409→          </label>
   410→          {error && <p className="rounded-lg bg-amber-50 px-3.5 py-2 text-[12.5px] font-semibold text-amber-700">{error}</p>}
   411→          <div className="flex justify-end gap-2.5 pt-1">
   412→            <button onClick={onClose} className="rounded-lg border border-slate-200 px-4 py-2.5 text-sm font-semibold text-slate-700 transition-colors hover:bg-slate-50">
   413→              {t("cancel")}
   414→            </button>
   415→            <button
   416→              onClick={submit}
   417→              disabled={busy}
   418→              className="rounded-lg bg-emerald-600 px-4 py-2.5 text-sm font-semibold text-white transition-colors hover:bg-emerald-700 disabled:opacity-50"
   419→            >
   420→              {t("requestSubmit")}
   421→            </button>
   422→          </div>
   423→        </div>
   424→      </div>
   425→    </div>
   426→  );
   427→}

   478→/* ---------------- profile & account ---------------- */
   479→
   480→export function ProfileCard({
   481→  view,
   482→  t,
   483→  onProfileSave,
   484→  onPasswordSave,
   485→  lang,
   486→}: {
   487→  view: ClientView;
   488→  t: T;
   489→  onProfileSave: (patch: { phone: string; country: string; address: string; city: string; postcode: string }) => Promise<string | null>;
   490→  onPasswordSave: (current: string, next: string) => Promise<string | null>;
   491→  lang: Lang;
   492→}) {
   493→  const c = view.client;
   494→  const fin = view.financials;
   495→  const [editing, setEditing] = useState(false);
   496→  const [form, setForm] = useState({ phone: c.phone, country: c.country, address: c.address, city: c.city, postcode: c.postcode });
   497→  const [busy, setBusy] = useState(false);
   498→  const [pwOpen, setPwOpen] = useState(false);
   499→  const [pw, setPw] = useState({ current: "", next: "", confirm: "" });
   500→  const [pwError, setPwError] = useState<string | null>(null);
   501→
   502→  const save = async () => {
   503→    setBusy(true);
   504→    const err = await onProfileSave(form);
   505→    setBusy(false);
   506→    if (!err) setEditing(false);
   507→  };
   508→
   509→  const changePassword = async () => {
   510→    if (pw.next !== pw.confirm) {
   511→      setPwError(t("passwordsMismatch"));
   512→      return;
   513→    }
   514→    setBusy(true);
   515→    const err = await onPasswordSave(pw.current, pw.next);
   516→    setBusy(false);
   517→    if (err) {
   518→      setPwError(t(err as StringKey));
   519→      return;
   520→    }
   521→    setPwOpen(false);
   522→    setPw({ current: "", next: "", confirm: "" });
   523→    setPwError(null);
   524→  };
   525→
   526→  const rows: Array<[string, string]> = [
   527→    [t("emailAddress"), c.email],
   528→    [t("phone"), c.phone || "—"],
   529→    [t("address"), [c.address, c.city, c.postcode].filter(Boolean).join(", ") || "—"],
   530→    [t("country"), c.country || "—"],
   531→    [t("accountNo"), c.accountNo],
   532→    [t("openingBalance"), usd(c.openingBalance)],
   533→    [t("creditsLabel"), usd(fin.credits)],
   534→    [t("debitsLabel"), usd(fin.debits)],
   535→    [t("statusLabel"), c.status === "active" ? t("active") : t("suspended")],
   536→    [t("tierLabel"), c.tier === "Private" ? t("tierPrivate") : c.tier === "Premium" ? t("tierPremium") : t("tierStandard")],
   537→  ];
   538→
   539→  return (
   540→    <Card className="overflow-hidden">
   541→      <SectionTitle
   542→        title={t("profileTitle")}
   543→        right={
   544→          !editing ? (
   545→            <button
   546→              onClick={() => {
   547→                setForm({ phone: c.phone, country: c.country, address: c.address, city: c.city, postcode: c.postcode });
   548→                setEditing(true);
   549→              }}
   550→              className="inline-flex items-center gap-1.5 rounded-lg border border-slate-200 px-2.5 py-1.5 text-[12px] font-semibold text-slate-600 transition-colors hover:bg-slate-50"
   551→            >
   552→              <Pencil className="h-3.5 w-3.5" /> {t("editProfile")}
   553→            </button>
   554→          ) : undefined
   555→        }
   556→      />
   557→      <div className="px-5 pb-5 pt-3 sm:px-6">
   558→        {editing ? (
   559→          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   560→            <label className="block">
   561→              <span className="mb-1.5 block text-[13px] font-medium text-slate-600">{t("phone")}</span>
   562→              <input value={form.phone} onChange={(e) => setForm({ ...form, phone: e.target.value })} dir="ltr" className="h-11 w-full rounded-lg border border-slate-200 px-3.5 text-sm text-slate-900 outline-none focus:border-emerald-500 focus:ring-2 focus:ring-emerald-500/15" />
   563→            </label>
   564→            <label className="block">
   565→              <span className="mb-1.5 block text-[13px] font-medium text-slate-600">{t("country")}</span>
   566→              <input value={form.country} onChange={(e) => setForm({ ...form, country: e.target.value })} className="h-11 w-full rounded-lg border border-slate-200 px-3.5 text-sm text-slate-900 outline-none focus:border-emerald-500 focus:ring-2 focus:ring-emerald-500/15" />
   567→            </label>
   568→            <label className="block sm:col-span-2">
   569→              <span className="mb-1.5 block text-[13px] font-medium text-slate-600">{t("address")}</span>
   570→              <input value={form.address} onChange={(e) => setForm({ ...form, address: e.target.value })} className="h-11 w-full rounded-lg border border-slate-200 px-3.5 text-sm text-slate-900 outline-none focus:border-emerald-500 focus:ring-2 focus:ring-emerald-500/15" />
   571→            </label>
   572→            <label className="block">
   573→              <span className="mb-1.5 block text-[13px] font-medium text-slate-600">{t("city")}</span>
   574→              <input value={form.city} onChange={(e) => setForm({ ...form, city: e.target.value })} className="h-11 w-full rounded-lg border border-slate-200 px-3.5 text-sm text-slate-900 outline-none focus:border-emerald-500 focus:ring-2 focus:ring-emerald-500/15" />
   575→            </label>
   576→            <label className="block">
   577→              <span className="mb-1.5 block text-[13px] font-medium text-slate-600">{t("postcode")}</span>
   578→              <input value={form.postcode} onChange={(e) => setForm({ ...form, postcode: e.target.value })} className="h-11 w-full rounded-lg border border-slate-200 px-3.5 text-sm text-slate-900 outline-none focus:border-emerald-500 focus:ring-2 focus:ring-emerald-500/15" />
   579→            </label>
   580→            <div className="flex justify-end gap-2.5 sm:col-span-2">
   581→              <button onClick={() => setEditing(false)} className="rounded-lg border border-slate-200 px-4 py-2.5 text-sm font-semibold text-slate-700 transition-colors hover:bg-slate-50">
   582→                {t("cancel")}
   583→              </button>
   584→              <button onClick={save} disabled={busy} className="rounded-lg bg-emerald-600 px-4 py-2.5 text-sm font-semibold text-white transition-colors hover:bg-emerald-700 disabled:opacity-50">
   585→                {t("saveProfile")}
   586→              </button>
   587→            </div>
   588→          </div>
   589→        ) : (
   590→          <dl className="grid grid-cols-1 gap-x-8 sm:grid-cols-2">
   591→            {rows.map(([k, v]) => (
   592→              <div key={k} className="flex items-center justify-between gap-4 border-b border-slate-50 py-2.5 last:border-0">
   593→                <dt className="shrink-0 text-[12.5px] font-medium text-slate-400">{k}</dt>
   594→                <dd className="truncate text-[13px] font-semibold text-slate-900" dir={k === t("emailAddress") || k === t("phone") || k === t("accountNo") || v.match(/^[+$\d]/) ? "ltr" : undefined}>
   595→                  {v}
   596→                </dd>
   597→              </div>
   598→            ))}
   599→          </dl>
   600→        )}
   601→
   602→        <div className="mt-5 border-t border-slate-100 pt-4">
   603→          {!pwOpen ? (
   604→            <button onClick={() => setPwOpen(true)} className="text-[13px] font-bold text-emerald-600 transition-colors hover:text-emerald-700">
   605→              {t("changePassword")}
   606→            </button>
   607→          ) : (
   608→            <div className="max-w-sm space-y-3">
   609→              <label className="block">
   610→                <span className="mb-1.5 block text-[13px] font-medium text-slate-600">{t("currentPassword")}</span>
   611→                <PasswordInput theme="light" value={pw.current} onChange={(v) => setPw({ ...pw, current: v })} autoComplete="current-password" />
   612→              </label>
   613→              <label className="block">
   614→                <span className="mb-1.5 block text-[13px] font-medium text-slate-600">{t("newPassword")}</span>
   615→                <PasswordInput theme="light" value={pw.next} onChange={(v) => setPw({ ...pw, next: v })} autoComplete="new-password" />
   616→              </label>
   617→              <label className="block">
   618→                <span className="mb-1.5 block text-[13px] font-medium text-slate-600">{t("confirmNewPassword")}</span>
   619→                <PasswordInput theme="light" value={pw.confirm} onChange={(v) => setPw({ ...pw, confirm: v })} autoComplete="new-password" />
   620→              </label>
   621→              {pwError && <p className="rounded-lg bg-amber-50 px-3.5 py-2 text-[12.5px] font-semibold text-amber-700">{pwError}</p>}
   622→              <div className="flex justify-end gap-2.5">
   623→                <button
   624→                  onClick={() => {
   625→                    setPwOpen(false);
   626→                    setPwError(null);
   627→                  }}
   628→                  className="rounded-lg border border-slate-200 px-4 py-2.5 text-sm font-semibold text-slate-700 transition-colors hover:bg-slate-50"
   629→                >
   630→                  {t("cancel")}
   631→                </button>
   632→                <button onClick={changePassword} disabled={busy} className="rounded-lg bg-emerald-600 px-4 py-2.5 text-sm font-semibold text-white transition-colors hover:bg-emerald-700 disabled:opacity-50">
   633→                  {t("changePassword")}
   634→                </button>
   635→              </div>
   636→            </div>
   637→          )}
   638→        </div>
   639→      </div>
   640→      {lang === "ar" && <span className="hidden">{c.name}</span>}
   641→    </Card>
   642→  );
   643→}
