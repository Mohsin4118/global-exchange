   275→/* ---------------- formatting helpers ---------------- */
   276→
   277→export function usd(n: number, opts?: { signed?: boolean; cents?: boolean }): string {
   278→  const cents = opts?.cents ?? true;
   279→  const abs = Math.abs(n);
   280→  const str = abs.toLocaleString("en-US", {
   281→    minimumFractionDigits: cents ? 2 : 0,
   282→    maximumFractionDigits: cents ? 2 : 0,
   283→  });
   284→  const sign = opts?.signed ? (n < 0 ? "-" : "+") : n < 0 ? "-" : "";
   285→  return `${sign}$${str}`;
   286→}
   287→
   288→export function cents(n: number): string {
   289→  return n.toLocaleString("en-US", { minimumFractionDigits: 2, maximumFractionDigits: 2 });
   290→}
