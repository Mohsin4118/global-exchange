    66→export function ClientDashboard({
    67→  t,
    68→  lang,
    69→  coins,
    70→  stocks,
    71→  onLangToggle,
    72→  onSignOut,
    73→}: {
    74→  t: (k: StringKey) => string;
    75→  lang: Lang;
    76→  coins: Coin[];
    77→  stocks: Coin[];
    78→  onLangToggle: () => void;
    79→  onSignOut: () => void;
    80→}) {
    81→  const { toast } = useToast();
    82→  const [view, setView] = useState<ClientView | null>(null);
    83→  const [ready, setReady] = useState(false);
    84→  const [section, setSection] = useState<Section>("overview");
    85→  const [drawer, setDrawer] = useState(false);
    86→  const [txFilter, setTxFilter] = useState<TxFilter>("all");
    87→  const [reqModal, setReqModal] = useState<null | RequestKind>(null);
    88→  const [showAllTxs, setShowAllTxs] = useState(false);
    89→
    90→  /* ---- live self-refresh: re-read the authorized API snapshot ---- */
    91→  const load = useCallback(async () => {
    92→    const res = await apiClientGet();
    93→    if (!res.ok) {
    94→      if (res.error === "unauthorized" || res.error === "gone") {
    95→        toast({ title: t("accountClosedToast") });
    96→        onSignOut();
    97→      }
    98→      return;
    99→    }
   100→    setView({ client: res.client!, financials: res.financials!, txs: res.txs!, notifications: res.notifications!, fx: res.fx! });
   101→    setReady(true);
   102→  }, [onSignOut, t, toast]);
   103→
   104→  useEffect(() => {
   105→    const boot = () => {
   106→      void load();
   107→    };
   108→    boot();
   109→    const id = setInterval(boot, 4000);
   110→    window.addEventListener("focus", boot);
   111→    return () => {
   112→      clearInterval(id);
   113→      window.removeEventListener("focus", boot);
   114→    };
   115→  }, [load]);
   116→
   117→  /* ---- live portfolio math (server balance + live market prices) ---- */
   118→  const markets = useMemo(() => [...coins, ...stocks], [coins, stocks]);
   119→  const holdingRows = useMemo(() => (view ? buildHoldingRows(view.client.holdings, markets) : []), [view, markets]);
   120→  const invested = holdingRows.reduce((s, r) => s + r.value, 0);
   121→  const cash = view?.financials.balance ?? 0;
   122→  const available = view?.financials.available ?? 0;
   123→  const total = cash + invested;
   124→  const portfolioChange = invested > 0 ? holdingRows.reduce((s, r) => s + r.coin.change24h * r.value, 0) / invested : 0;
   125→  const totalDeposits = view?.financials.credits ?? 0;
   126→  const totalWithdrawn = useMemo(() => (view ? view.txs.filter((x) => x.kind === "withdrawal" && x.status === "COMPLETED").reduce((s, x) => s + x.amount, 0) : 0), [view]);
   127→
   128→  const suspended = view?.client.status === "suspended";
   129→  const unread = view?.notifications.filter((n) => n.unread).length ?? 0;
   130→
   131→  /* ---- display currency: one formatter drives every figure (#10) ---- */
   132→  const cur = view?.client.currency ?? "USD";
   133→  const fx = view?.fx ?? { USD: 1 };
   134→  const money = useMemo(() => makeMoney(cur, fx), [cur, fx]);
   135→  const setCurrency = async (code: CurrencyCode) => {
   136→    if (!CURRENCIES.includes(code)) return;
   137→    const res = await apiClientAction({ action: "set-currency", currency: code });
   138→    if (res.ok && res.client && res.txs) {
   139→      setView({ client: res.client, financials: res.financials!, txs: res.txs, notifications: res.notifications!, fx: res.fx! });
   140→    }
   141→  };
   142→
   143→  /* ---- session performance chart ---- */
   144→  const [history, setHistory] = useState<number[]>([]);
   145→  const lastTotalRef = useRef(total);
   146→  useEffect(() => {
   147→    setHistory((h) => {
   148→      if (h.length > 0 || total <= 0) return h;
   149→      const pts: number[] = [];
   150→      let v = total * 0.982;
   151→      let s = Math.max(1, Math.floor(total)) % 233280 || 13;
   152→      const rand = () => {
   153→        s = (s * 9301 + 49297) % 233280;
   154→        return s / 233280;
   155→      };
   156→      for (let i = 0; i < 42; i++) {
   157→        v += total * (rand() - 0.47) * 0.004;
   158→        pts.push(Math.max(total * 0.9, v));
   159→      }
   160→      pts[pts.length - 1] = total;
   161→      lastTotalRef.current = total;
   162→      return pts;
   163→    });
   164→  }, [total]);
   165→
   166→  useEffect(() => {
   167→    const append = () => {
   168→      if (history.length === 0) return;
   169→      if (Math.abs(total - lastTotalRef.current) < 0.005) return;
   170→      lastTotalRef.current = total;
   171→      setHistory((h) => [...h.slice(-79), total]);
   172→    };
   173→    append();
   174→  }, [total, history.length]);
   175→
   176→  /* ---- actions ---- */
   177→  const submitRequest = async (payload: { kind: RequestKind; amount: number; asset: string; destination: string; note: string }): Promise<string | null> => {
   178→    const res = await apiClientAction({ action: "request-transaction", ...payload });
   179→    if (!res.ok) {
   180→      return res.error === "funds" ? "requestTooMuch" : res.error === "duplicate" ? "requestDuplicate" : "requestInvalid";
   181→    }
   182→    if (res.client && res.txs) setView({ client: res.client, financials: res.financials!, txs: res.txs, notifications: res.notifications!, fx: res.fx! });
   183→    toast({ title: t("requestSent"), description: t("requestSentSub") });
   184→    setReqModal(null);
   185→    return null;
   186→  };
   187→
   188→  /* the ONLY mutation a client may make on a request: cancelling their own PENDING one */
   189→  const cancelRequest = async (tx: { id: string; reference: string }) => {
   190→    const res = await apiClientAction({ action: "cancel-request", id: tx.id });
   191→    if (!res.ok) {
   192→      toast({ title: t("requestInvalid") });
   193→      return;
   194→    }
   195→    if (res.client && res.txs) setView({ client: res.client, financials: res.financials!, txs: res.txs, notifications: res.notifications!, fx: res.fx! });
   196→    toast({ title: t("requestCancelledToast"), description: t("requestCancelledSub") });
   197→  };
   198→
   199→  const markAllRead = async () => {
   200→    const res = await apiClientAction({ action: "mark-read", ids: "all" });
   201→    if (res.ok && res.client && res.txs) {
   202→      setView({ client: res.client, financials: res.financials!, txs: res.txs, notifications: res.notifications!, fx: res.fx! });
   203→    }
   204→  };
   205→
   206→  const saveProfile = async (patch: { phone: string; country: string; address: string; city: string; postcode: string }): Promise<string | null> => {
   207→    const res = await apiClientAction({ action: "update-profile", ...patch });
   208→    if (!res.ok) return "requestInvalid";
   209→    if (res.client && res.txs) setView({ client: res.client, financials: res.financials!, txs: res.txs, notifications: res.notifications!, fx: res.fx! });
   210→    toast({ title: t("profileSaved") });
   211→    return null;
   212→  };
   213→
   214→  const savePassword = async (current: string, next: string): Promise<string | null> => {
   215→    const res = await apiClientAction({ action: "change-password", current, next });
   216→    if (!res.ok) return res.error === "wrong-current" ? "wrongCurrentPassword" : res.error === "weak" ? "weakPassword" : "requestInvalid";
   217→    toast({ title: t("passwordChangedToast") });
   218→    return null;
   219→  };
   220→
   221→  const signOut = async () => {
   222→    await apiLogout("client");
   223→    onSignOut();
   224→  };
   225→
   226→  const downloadCsv = () => {
   227→    if (!view) return;
   228→    const header = "Date,Description,Asset,Method,Type,Amount USD,Status,Reference\n";
   229→    const lines = view.txs.map((tx) =>
   230→      [tx.dateISO, `"${(tx.labelKey ? t(tx.labelKey as StringKey) : tx.label).replace(/"/g, '""')}"`, tx.asset ?? "", `"${tx.method}"`, tx.type, tx.amount.toFixed(2), tx.status, tx.reference].join(","),
   231→    );
   232→    const blob = new Blob([header + lines.join("\n")], { type: "text/csv;charset=utf-8" });
   233→    const url = URL.createObjectURL(blob);
   234→    const a = document.createElement("a");
   235→    a.href = url;
   236→    a.download = `cryptowise-statement-${view.client.accountNo}.csv`;
   237→    a.click();
   238→    URL.revokeObjectURL(url);
   239→    toast({ title: t("statementToast") });
   240→  };
   241→
   242→  /* ---- derived lists ---- */
   243→  const txs = view?.txs ?? [];
   244→  const filteredTxs = useMemo(() => txs.filter((tx) => txFilter === "all" || (txFilter === "in" ? tx.type === "CREDIT" : tx.type === "DEBIT")), [txs, txFilter]);
   245→  const deposits = useMemo(() => txs.filter((tx) => tx.kind === "deposit"), [txs]);
   246→  const withdrawals = useMemo(() => txs.filter((tx) => tx.kind === "withdrawal"), [txs]);
   247→  const pendingWithdrawalTotal = withdrawals.filter((w) => w.status === "PENDING" || w.status === "PROCESSING").reduce((s, w) => s + w.amount, 0);
   248→
   249→  const allocation = useMemo(() => {
   250→    const segs = holdingRows.map((r) => ({ label: r.coin.name, value: r.value, color: r.coin.color }));
   251→    if (cash > 0) segs.unshift({ label: t("cash"), value: cash, color: "#10b981" });
   252→    return segs;
   253→  }, [holdingRows, cash, t]);
   254→
   255→  /* ---- menu definition (every item works — switches sections) ---- */
   256→  const NAV: Array<{ id: Section; label: string; icon: React.ReactNode; badge?: number }> = [
   257→    { id: "overview", label: t("menuOverview"), icon: <LayoutDashboard className="h-4 w-4" /> },
   258→    { id: "portfolio", label: t("menuPortfolio"), icon: <PieChart className="h-4 w-4" /> },
   259→    { id: "transactions", label: t("menuTransactions"), icon: <ArrowLeftRight className="h-4 w-4" /> },
   260→    { id: "deposits", label: t("menuDeposits"), icon: <ArrowDownLeft className="h-4 w-4" /> },
   261→    { id: "withdrawals", label: t("menuWithdrawals"), icon: <ArrowUpRight className="h-4 w-4" /> },
   262→    { id: "notifications", label: t("menuNotifications"), icon: <Bell className="h-4 w-4" />, badge: unread },
   263→    { id: "account", label: t("menuAccount"), icon: <UserRoundCog className="h-4 w-4" /> },
   264→  ];
   265→
   266→  const go = (s: Section) => {
   267→    setSection(s);
   268→    setDrawer(false);
   269→    window.scrollTo({ top: 0 });
   270→  };
   271→
   272→  if (!ready || !view) {
   273→    return (
   274→      <div className="flex min-h-screen items-center justify-center bg-[#f5f7fa]">
   275→        <div className="flex flex-col items-center gap-3">
   276→          <span className="h-9 w-9 animate-spin rounded-full border-[3px] border-emerald-200 border-t-emerald-600" />
   277→          <p className="text-sm font-semibold text-slate-400">CryptoWise</p>
   278→        </div>
   279→      </div>
   280→    );
   281→  }
   282→
   283→  const c = view.client;
   284→
   285→  return (
   286→    <div className="min-h-screen overflow-x-clip bg-[#f5f7fa] text-slate-900 [color-scheme:light]">
   287→
   288→      {/* ---------- topbar ---------- */}
   289→      <header className="sticky top-0 z-40 border-b border-slate-200/80 bg-white/90 backdrop-blur-xl">
   290→        <div className="mx-auto flex h-14 w-full max-w-[1280px] items-center justify-between gap-2 px-4 sm:h-16 sm:gap-3 sm:px-6">
   291→          <div className="flex min-w-0 items-center gap-2.5 sm:gap-3">
   292→            <button
   293→              onClick={() => setDrawer(true)}
   294→              aria-label="Open menu"
   295→              aria-expanded={drawer}
   296→              className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg border border-slate-200 text-slate-600 transition-colors hover:bg-slate-100 hover:text-slate-900 lg:hidden"
   297→            >
   298→              <Menu className="h-4.5 w-4.5" />
   299→            </button>
   300→            <LogoMark className="h-8 w-8 shrink-0 sm:h-9 sm:w-9" />
   301→            <div className="min-w-0">
   302→              <p className="truncate text-[13.5px] font-bold leading-tight text-slate-900 sm:text-[14px]">CryptoWise</p>
   303→              <p className="truncate text-[10.5px] font-semibold text-emerald-600 sm:text-[11px]">{t("clientArea")}</p>
   304→            </div>
   305→            <span className="ms-1 hidden shrink-0 rounded-full bg-slate-100 px-2.5 py-1 text-[11px] font-bold text-slate-500 ring-1 ring-white/10 md:inline-block" dir="ltr">
   306→              {t("accountNo")} {c.accountNo}
   307→            </span>
   308→          </div>
   309→          <div className="flex shrink-0 items-center gap-1.5 sm:gap-2">
   310→            <button
   311→              onClick={onLangToggle}
   312→              className="flex h-9 items-center rounded-lg border border-slate-200 px-2.5 text-[11.5px] font-bold text-slate-600 transition-colors hover:bg-slate-100 hover:text-slate-900 sm:px-3 sm:text-[12px]"
   313→              aria-label="Toggle language"
   314→            >
   315→              {lang === "en" ? "العربية" : "EN"}
   316→            </button>
   317→            <button
   318→              onClick={() => go("notifications")}
   319→              aria-label={t("notificationsTitle")}
   320→              className="relative flex h-9 w-9 items-center justify-center rounded-lg border border-slate-200 text-slate-600 transition-colors hover:bg-slate-100 hover:text-slate-900"
   321→            >
   322→              <Bell className="h-4 w-4" />
   323→              {unread > 0 && (
   324→                <span className="absolute -right-1 -top-1 flex h-4 min-w-4 items-center justify-center rounded-full bg-amber-400 px-1 text-[9.5px] font-bold text-[#022c20]">{unread}</span>
   325→              )}
   326→            </button>
   327→            <div className="hidden items-center gap-2.5 rounded-full border border-slate-200 py-1 pe-3 ps-1 sm:flex">
   328→              <span className="flex h-7 w-7 items-center justify-center rounded-full bg-[#00E5A0] text-[11px] font-bold text-[#022c20]">{initials(c.name)}</span>
   329→              <span className="hidden text-left leading-tight md:block" dir="auto">
   330→                <span className="block max-w-[120px] truncate text-[12px] font-bold text-slate-900">{c.name}</span>
   331→                <span className="block text-[10.5px] font-semibold text-slate-500">{t("privateClient")}</span>
   332→              </span>
   333→            </div>
   334→            <button
   335→              onClick={signOut}
   336→              aria-label={t("signOut")}
   337→              className="flex h-9 items-center gap-1.5 rounded-lg border border-slate-200 px-2.5 text-[12px] font-semibold text-slate-600 transition-colors hover:bg-slate-100 hover:text-slate-900 sm:px-3"
   338→            >
   339→              <LogOut className="h-3.5 w-3.5" />
   340→              <span className="hidden md:inline">{t("signOut")}</span>
   341→            </button>
   342→          </div>
   343→        </div>
   344→      </header>
   345→
   346→      {/* ---------- body: sidebar + content ---------- */}
   347→      <div className="relative mx-auto flex w-full max-w-[1280px] items-start gap-6 px-4 sm:px-6">
   348→        {/* desktop sidebar */}
   349→        <aside className="sticky top-[72px] hidden w-[218px] shrink-0 py-6 lg:block" aria-label="Client menu">
   350→          <nav className="overflow-hidden rounded-2xl border border-slate-200/80 bg-white shadow-sm p-2">
   351→            {NAV.map((item) => (
   352→              <button
   353→                key={item.id}
   354→                onClick={() => go(item.id)}
   355→                aria-current={section === item.id ? "page" : undefined}
   356→                className={cn(
   357→                  "flex w-full items-center gap-2.5 rounded-xl px-3 py-2.5 text-[13px] font-semibold transition-colors",
   358→                  section === item.id ? "bg-emerald-50 text-emerald-600" : "text-slate-600 hover:bg-slate-50 hover:text-slate-900",
   359→                )}
   360→              >
   361→                {item.icon}
   362→                <span className="flex-1 text-start">{item.label}</span>
   363→                {!!item.badge && item.badge > 0 && (
   364→                  <span className="flex h-4.5 min-w-[18px] items-center justify-center rounded-full bg-amber-400 px-1 text-[10px] font-bold text-[#022c20]">{item.badge}</span>
   365→                )}
   366→              </button>
   367→            ))}
   368→            <div className="my-2 h-px bg-slate-100" />
   369→            <button
   370→              onClick={signOut}
   371→              className="flex w-full items-center gap-2.5 rounded-xl px-3 py-2.5 text-[13px] font-semibold text-slate-600 transition-colors hover:bg-slate-50 hover:text-slate-900"
   372→            >
   373→              <LogOut className="h-4 w-4" />
   374→              <span className="flex-1 text-start">{t("signOut")}</span>
   375→            </button>
   376→          </nav>
   377→          <p className="mt-3 px-3 text-[10.5px] leading-relaxed text-slate-400" dir="ltr">
   378→            {t("accountNo")} {c.accountNo}
   379→          </p>
   380→        </aside>
   381→
   382→        {/* content */}
   383→        <main className="min-w-0 flex-1 py-5 sm:py-6">
   384→          {section === "overview" && (
   385→            <OverviewSection
   386→              view={view}
   387→              t={t}
   388→              lang={lang}
   389→              coins={coins}
   390→              stocks={stocks}
   391→              holdingRows={holdingRows}
   392→              invested={invested}
   393→              cash={cash}
   394→              total={total}
   395→              available={available}
   396→              portfolioChange={portfolioChange}
   397→              totalDeposits={totalDeposits}
   398→              pendingWithdrawalTotal={pendingWithdrawalTotal}
   399→              history={history}
   400→              allocation={allocation}
   401→              suspended={!!suspended}
   402→              money={money}
   403→              cur={cur}
   404→              onCurrency={setCurrency}
   405→              onDeposit={() => !suspended && setReqModal("deposit")}
   406→              onWithdraw={() => !suspended && setReqModal("withdrawal")}
   407→              onGo={go}
   408→            />
   409→          )}
   410→
   411→          {section === "portfolio" && (
   412→            <div className="grid grid-cols-1 gap-5">
   413→              <SectionHeading title={t("menuPortfolio")} sub={t("welcomeSub")} />
   414→              <HoldingsTable rows={holdingRows} t={t} lang={lang} />
   415→              <AllocationBar segments={allocation} t={t} />
   416→              <MarketStrip coins={coins} stocks={stocks} t={t} />
   417→            </div>
   418→          )}
   419→
   420→          {section === "transactions" && (
   421→            <div className="grid grid-cols-1 gap-5">
   422→              <SectionHeading title={t("txHistory")} sub={`${txs.length} · ${t("allTime")}`} />
   423→              <div className="flex flex-wrap items-center justify-between gap-2.5">
   424→                <div className="flex items-center gap-1 rounded-lg bg-slate-100 p-1">
   425→                  {(
   426→                    [
   427→                      ["all", t("filterAll")],
   428→                      ["in", t("filterIn")],
   429→                      ["out", t("filterOut")],
   430→                    ] as const
   431→                  ).map(([key, label]) => (
   432→                    <button
   433→                      key={key}
   434→                      onClick={() => setTxFilter(key)}
   435→                      className={cn("rounded-md px-2.5 py-1.5 text-[11.5px] font-bold transition-colors", txFilter === key ? "bg-emerald-100 text-emerald-600" : "text-slate-500 hover:text-slate-900")}
   436→                    >
   437→                      {label}
   438→                    </button>
   439→                  ))}
   440→                </div>
   441→                <div className="flex items-center gap-2">
   442→                  <button
   443→                    onClick={() => !suspended && setReqModal("deposit")}
   444→                    disabled={suspended}
