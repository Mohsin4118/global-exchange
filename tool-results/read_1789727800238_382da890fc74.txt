   242→function NewClientModal({ open, onClose, ctx }: { open: boolean; onClose: () => void; ctx: AdminCtx }) {
   243→  const [name, setName] = useState("");
   244→  const [email, setEmail] = useState("");
   245→  const [password, setPassword] = useState("");
   246→  const [phone, setPhone] = useState("");
   247→  const [country, setCountry] = useState("");
   248→  const [address, setAddress] = useState("");
   249→  const [balance, setBalance] = useState("0.00");
   250→  const [busy, setBusy] = useState(false);
   251→
   252→  const submit = async () => {
   253→    if (!name.trim() || !email.trim() || !password.trim()) {
   254→      ctx.toast("Missing information", "Name, email and portal password are required.");
   255→      return;
   256→    }
   257→    if (password.trim().length < 6) {
   258→      ctx.toast("Weak password", "The portal password must be at least 6 characters.");
   259→      return;
   260→    }
   261→    setBusy(true);
   262→    const ok = await ctx.runAction(
   263→      {
   264→        action: "create-client",
   265→        client: { name: name.trim(), email: email.trim(), password: password.trim(), phone: phone.trim(), country: country.trim(), address: address.trim(), openingBalance: parseFloat(balance) || 0 },
   266→      },
   267→      {
   268→        title: "Client created",
   269→        description: `${name.trim()} was added. Portal access is live — ${email.trim()} can sign in and open their dashboard.`,
   270→      },
   271→    );
   272→    setBusy(false);
   273→    if (ok) {
   274→      setName(""); setEmail(""); setPassword(""); setPhone(""); setCountry(""); setAddress(""); setBalance("0.00");
   275→      onClose();
   276→    }
   277→  };
   278→
   279→  return (
   280→    <Modal open={open} onClose={onClose} title="New Client">
   281→      <div className="space-y-4">
   282→        <TextInput label="Full Name *" value={name} onChange={setName} placeholder="e.g. Charlie Williams" />
   283→        <TextInput label="Email *" type="email" value={email} onChange={setEmail} placeholder="client@email.com" />
   284→        <div>
   285→          <span className="mb-1.5 block text-[13px] font-medium text-slate-600">Portal Password *</span>
   286→          <PasswordInput theme="light" value={password} onChange={setPassword} placeholder="Client sign-in password (min. 6 chars)" autoComplete="new-password" />
   287→        </div>
   288→        <p className="rounded-lg border border-emerald-100 bg-emerald-50/60 px-3.5 py-2.5 text-[12.5px] leading-relaxed text-slate-500">
   289→          The client signs in on the website with this email and password — their dashboard opens with the opening balance below.
   290→        </p>
   291→        <div className="grid grid-cols-2 gap-4">
   292→          <TextInput label="Phone" value={phone} onChange={setPhone} placeholder="Optional" />
   293→          <TextInput label="Country" value={country} onChange={setCountry} placeholder="Optional" />
   294→        </div>
   295→        <TextInput label="Address" value={address} onChange={setAddress} placeholder="Optional" />
   296→        <TextInput label="Opening Balance (USD)" type="number" value={balance} onChange={setBalance} />
   297→        <div className="flex justify-end gap-3 pt-2">
   298→          <OutlineButton onClick={onClose}>Cancel</OutlineButton>
   299→          <PrimaryButton onClick={submit} disabled={busy}>
   300→            Create Client
   301→          </PrimaryButton>
   302→        </div>
   303→      </div>
   304→    </Modal>
   305→  );
   306→}

   839→      {/* Comments */}
   840→      <Card className="mt-6 p-6">
   841→        <div className="flex items-center gap-2.5">
   842→          <MessageSquare className="h-5 w-5 text-slate-500" />
   843→          <h2 className="text-lg font-bold text-slate-900">Comments</h2>
   844→        </div>
   845→        <div className="mt-5 flex items-end gap-3">
   846→          <textarea
   847→            value={note}
   848→            onChange={(e) => setNote(e.target.value)}
   849→            placeholder="Add a note about this client..."
   850→            rows={2}
   851→            className="w-full resize-y rounded-lg border border-slate-200 px-4 py-3 text-sm outline-none transition-colors placeholder:text-slate-400 focus:border-emerald-500 focus:ring-2 focus:ring-emerald-500/15"
   852→          />
   853→          <PrimaryButton
   854→            className="mb-0.5 shrink-0 px-5 disabled:bg-emerald-600/40"
   855→            disabled={!note.trim()}
   856→            onClick={async () => {
   857→              const ok = await ctx.runAction({ action: "add-comment", clientId: client.id, body: note.trim() }, { title: "Comment posted" });
   858→              if (ok) setNote("");
   859→            }}
   860→          >
   861→            Post
   862→          </PrimaryButton>
   863→        </div>
   864→        {comments.length === 0 ? (
   865→          <div className="flex flex-col items-center py-12 text-center">
   866→            <span className="flex h-14 w-14 items-center justify-center rounded-full bg-slate-100">
   867→              <Inbox className="h-6 w-6 text-slate-400" />
   868→            </span>
   869→            <p className="mt-4 text-[15px] font-bold text-slate-900">No comments yet</p>
   870→            <p className="mt-1 text-sm text-slate-500">Notes left by staff about this client will appear here.</p>
   871→          </div>
   872→        ) : (
   873→          <ul className="mt-6 space-y-4">
   874→            {comments.map((c) => (
   875→              <li key={c.id} className="rounded-lg border border-slate-100 bg-slate-50/60 p-4">
   876→                <div className="flex items-center justify-between">
   877→                  <p className="text-sm font-semibold text-slate-900">{c.author}</p>
   878→                  <span className="text-xs text-slate-400">{c.time}</span>
   879→                </div>
   880→                <p className="mt-1.5 text-sm text-slate-600">{c.body}</p>
   881→              </li>
   882→            ))}
   883→          </ul>
   884→        )}
   885→      </Card>
   886→
   887→      {/* delete transaction confirmation */}
   888→      <Modal open={confirmDeleteTx !== null} onClose={() => setConfirmDeleteTx(null)} title="Delete Transaction">
   889→        <p className="text-sm leading-relaxed text-slate-600">
   890→          Delete this transaction permanently? The client&apos;s balance, statement, all dashboard statistics and reports will be recalculated immediately. This cannot be undone.
   891→        </p>
   892→        <div className="mt-5 flex justify-end gap-3">
   893→          <OutlineButton onClick={() => setConfirmDeleteTx(null)}>Cancel</OutlineButton>
   894→          <PrimaryButton className="!bg-amber-600 hover:!bg-amber-700" onClick={() => confirmDeleteTx && removeTx(confirmDeleteTx)}>
   895→            Delete Transaction
   896→          </PrimaryButton>
   897→        </div>
   898→      </Modal>
   899→
   900→      <span className="hidden"><X /></span>
   901→    </div>
   902→  );
   903→}
