    61→
    62→export function ClientDashboard({
    63→  t,
    64→  lang,
    65→  coins,
    66→  stocks,
    67→  onLangToggle,
    68→  onSignOut,
    69→}: {
    70→  t: (k: StringKey) => string;
    71→  lang: Lang;
    72→  coins: Coin[];
    73→  stocks: Coin[];
    74→  onLangToggle: () => void;
    75→  onSignOut: () => void;
    76→}) {
    77→  const { toast } = useToast();
    78→  const [view, setView] = useState<ClientView | null>(null);
    79→  const [ready, setReady] = useState(false);
    80→  const [section, setSection] = useState<Section>("overview");
    81→  const [drawer, setDrawer] = useState(false);
    82→  const [txFilter, setTxFilter] = useState<TxFilter>("all");
    83→  const [reqModal, setReqModal] = useState<null | "deposit" | "withdrawal">(null);
    84→  const [showAllTxs, setShowAllTxs] = useState(false);
    85→
    86→  /* ---- live self-refresh: re-read the authorized API snapshot ---- */
    87→  const load = useCallback(async () => {
    88→    const res = await apiClientGet();
    89→    if (!res.ok) {
    90→      if (res.error === "unauthorized" || res.error === "gone") {
    91→        toast({ title: t("accountClosedToast") });
    92→        onSignOut();
    93→      }
    94→      return;
    95→    }
    96→    setView({ client: res.client!, financials: res.financials!, txs: res.txs!, notifications: res.notifications! });
    97→    setReady(true);
    98→  }, [onSignOut, t, toast]);
    99→
   100→  useEffect(() => {
   101→    const boot = () => {
   102→      void load();
   103→    };
   104→    boot();
   105→    const id = setInterval(boot, 4000);
   106→    window.addEventListener("focus", boot);
   107→    return () => {
   108→      clearInterval(id);
   109→      window.removeEventListener("focus", boot);
   110→    };
   111→  }, [load]);
   112→
   113→  /* ---- live portfolio math (server balance + live market prices) ---- */
   114→  const markets = useMemo(() => [...coins, ...stocks], [coins, stocks]);
   115→  const holdingRows = useMemo(() => (view ? buildHoldingRows(view.client.holdings, markets) : []), [view, markets]);
   116→  const invested = holdingRows.reduce((s, r) => s + r.value, 0);
   117→  const cash = view?.financials.balance ?? 0;
   118→  const available = view?.financials.available ?? 0;
   119→  const total = cash + invested;
   120→  const portfolioChange = invested > 0 ? holdingRows.reduce((s, r) => s + r.coin.change24h * r.value, 0) / invested : 0;
   121→  const totalDeposits = view?.financials.credits ?? 0;
   122→  const totalWithdrawn = useMemo(() => (view ? view.txs.filter((x) => x.kind === "withdrawal" && x.status === "COMPLETED").reduce((s, x) => s + x.amount, 0) : 0), [view]);
   123→
   124→  const suspended = view?.client.status === "suspended";
   125→  const unread = view?.notifications.filter((n) => n.unread).length ?? 0;
   126→
   127→  /* ---- session performance chart ---- */
   128→  const [history, setHistory] = useState<number[]>([]);
   129→  const lastTotalRef = useRef(total);
   130→  useEffect(() => {
   131→    setHistory((h) => {
   132→      if (h.length > 0 || total <= 0) return h;
   133→      const pts: number[] = [];
   134→      let v = total * 0.982;
   135→      let s = Math.max(1, Math.floor(total)) % 233280 || 13;
   136→      const rand = () => {
   137→        s = (s * 9301 + 49297) % 233280;
   138→        return s / 233280;
   139→      };
   140→      for (let i = 0; i < 42; i++) {
   141→        v += total * (rand() - 0.47) * 0.004;
   142→        pts.push(Math.max(total * 0.9, v));
   143→      }
   144→      pts[pts.length - 1] = total;
   145→      lastTotalRef.current = total;
   146→      return pts;
   147→    });
   148→  }, [total]);
   149→
   150→  useEffect(() => {
   151→    const append = () => {
   152→      if (history.length === 0) return;
   153→      if (Math.abs(total - lastTotalRef.current) < 0.005) return;
   154→      lastTotalRef.current = total;
   155→      setHistory((h) => [...h.slice(-79), total]);
   156→    };
   157→    append();
   158→  }, [total, history.length]);
   159→
   160→  /* ---- actions ---- */
   161→  const submitRequest = async (kind: "deposit" | "withdrawal", amount: number, note: string): Promise<string | null> => {
   162→    const res = await apiClientAction({ action: "request-transaction", kind, amount, note });
   163→    if (!res.ok) {
   164→      return res.error === "funds" ? "requestTooMuch" : "requestInvalid";
   165→    }
   166→    if (res.client && res.txs) setView({ client: res.client, financials: res.financials!, txs: res.txs, notifications: res.notifications! });
   167→    toast({ title: t("requestSent"), description: t("requestSentSub") });
   168→    setReqModal(null);
   169→    return null;
   170→  };
   171→
   172→  const markAllRead = async () => {
   173→    const res = await apiClientAction({ action: "mark-read", ids: "all" });
   174→    if (res.ok && res.client && res.txs) {
   175→      setView({ client: res.client, financials: res.financials!, txs: res.txs, notifications: res.notifications! });
   176→    }
   177→  };
   178→
   179→  const saveProfile = async (patch: { phone: string; country: string; address: string; city: string; postcode: string }): Promise<string | null> => {
   180→    const res = await apiClientAction({ action: "update-profile", ...patch });
   181→    if (!res.ok) return "requestInvalid";
   182→    if (res.client && res.txs) setView({ client: res.client, financials: res.financials!, txs: res.txs, notifications: res.notifications! });
   183→    toast({ title: t("profileSaved") });
   184→    return null;
   185→  };
   186→
   187→  const savePassword = async (current: string, next: string): Promise<string | null> => {
   188→    const res = await apiClientAction({ action: "change-password", current, next });
   189→    if (!res.ok) return res.error === "wrong-current" ? "wrongCurrentPassword" : res.error === "weak" ? "weakPassword" : "requestInvalid";
   190→    toast({ title: t("passwordChangedToast") });
   191→    return null;
   192→  };
   193→
   194→  const signOut = async () => {
   195→    await apiLogout("client");
   196→    onSignOut();
   197→  };
   198→
   199→  const downloadCsv = () => {
   200→    if (!view) return;
   201→    const header = "Date,Description,Asset,Method,Type,Amount USD,Status,Reference\n";
   202→    const lines = view.txs.map((tx) =>
   203→      [tx.dateISO, `"${(tx.labelKey ? t(tx.labelKey as StringKey) : tx.label).replace(/"/g, '""')}"`, tx.asset ?? "", `"${tx.method}"`, tx.type, tx.amount.toFixed(2), tx.status, tx.reference].join(","),
   204→    );
   205→    const blob = new Blob([header + lines.join("\n")], { type: "text/csv;charset=utf-8" });
   206→    const url = URL.createObjectURL(blob);
   207→    const a = document.createElement("a");
   208→    a.href = url;
   209→    a.download = `cryptowise-statement-${view.client.accountNo}.csv`;
   210→    a.click();
   211→    URL.revokeObjectURL(url);
   212→    toast({ title: t("statementToast") });
   213→  };
   214→
   215→  /* ---- derived lists ---- */
   216→  const txs = view?.txs ?? [];
   217→  const filteredTxs = useMemo(() => txs.filter((tx) => txFilter === "all" || (txFilter === "in" ? tx.type === "CREDIT" : tx.type === "DEBIT")), [txs, txFilter]);
   218→  const deposits = useMemo(() => txs.filter((tx) => tx.kind === "deposit"), [txs]);
   219→  const withdrawals = useMemo(() => txs.filter((tx) => tx.kind === "withdrawal"), [txs]);
   220→  const pendingWithdrawalTotal = withdrawals.filter((w) => w.status === "PENDING" || w.status === "PROCESSING").reduce((s, w) => s + w.amount, 0);
   221→
   222→  const allocation = useMemo(() => {
   223→    const segs = holdingRows.map((r) => ({ label: r.coin.name, value: r.value, color: r.coin.color }));
   224→    if (cash > 0) segs.unshift({ label: t("cash"), value: cash, color: "#10b981" });
   225→    return segs;
   226→  }, [holdingRows, cash, t]);
   227→
   228→  /* ---- menu definition (every item works — switches sections) ---- */
   229→  const NAV: Array<{ id: Section; label: string; icon: React.ReactNode; badge?: number }> = [
   230→    { id: "overview", label: t("menuOverview"), icon: <LayoutDashboard className="h-4 w-4" /> },
   231→    { id: "portfolio", label: t("menuPortfolio"), icon: <PieChart className="h-4 w-4" /> },
   232→    { id: "transactions", label: t("menuTransactions"), icon: <ArrowLeftRight className="h-4 w-4" /> },
   233→    { id: "deposits", label: t("menuDeposits"), icon: <ArrowDownLeft className="h-4 w-4" /> },
   234→    { id: "withdrawals", label: t("menuWithdrawals"), icon: <ArrowUpRight className="h-4 w-4" /> },
   235→    { id: "notifications", label: t("menuNotifications"), icon: <Bell className="h-4 w-4" />, badge: unread },
   236→    { id: "account", label: t("menuAccount"), icon: <UserRoundCog className="h-4 w-4" /> },
   237→  ];
   238→
   239→  const go = (s: Section) => {
   240→    setSection(s);
   241→    setDrawer(false);
   242→    window.scrollTo({ top: 0 });
   243→  };
   244→
   245→  if (!ready || !view) {
   246→    return (
   247→      <div className="flex min-h-screen items-center justify-center bg-[#04121c]">
   248→        <div className="flex flex-col items-center gap-3">
   249→          <span className="h-9 w-9 animate-spin rounded-full border-[3px] border-[#00E5A0]/25 border-t-[#00E5A0]" />
   250→          <p className="text-sm font-semibold text-white/40">CryptoWise</p>
   251→        </div>
   252→      </div>
   253→    );
   254→  }
   255→
   256→  const c = view.client;
   257→
   258→  return (
   259→    <div className="min-h-screen bg-[#04121c] text-white [color-scheme:dark]">
   260→      {/* ambient teal glows */}
   261→      <div className="pointer-events-none fixed inset-0" aria-hidden="true">
   262→        <div className="absolute -top-40 start-[10%] h-[420px] w-[420px] rounded-full bg-[#00E5A0]/[0.05] blur-[130px]" />
   263→        <div className="absolute bottom-[-20%] end-[-8%] h-[380px] w-[380px] rounded-full bg-teal-400/[0.04] blur-[120px]" />
   264→      </div>
   265→
   266→      {/* ---------- topbar ---------- */}
   267→      <header className="sticky top-0 z-40 border-b border-white/[0.06] bg-[#031019]/90 backdrop-blur-xl">
   268→        <div className="mx-auto flex h-14 w-full max-w-[1280px] items-center justify-between gap-2 px-3 sm:h-16 sm:gap-3 sm:px-6">
   269→          <div className="flex min-w-0 items-center gap-2.5 sm:gap-3">
   270→            <button
   271→              onClick={() => setDrawer(true)}
   272→              aria-label="Open menu"
   273→              aria-expanded={drawer}
   274→              className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg border border-white/10 text-white/60 transition-colors hover:bg-white/[0.05] hover:text-white lg:hidden"
   275→            >
   276→              <Menu className="h-4.5 w-4.5" />
   277→            </button>
   278→            <LogoMark className="h-8 w-8 shrink-0 sm:h-9 sm:w-9" />
   279→            <div className="min-w-0">
   280→              <p className="truncate text-[13.5px] font-bold leading-tight text-white sm:text-[14px]">CryptoWise</p>
   281→              <p className="truncate text-[10.5px] font-semibold text-[#00E5A0] sm:text-[11px]">{t("clientArea")}</p>
   282→            </div>
   283→            <span className="ms-1 hidden shrink-0 rounded-full bg-white/[0.06] px-2.5 py-1 text-[11px] font-bold text-white/50 ring-1 ring-white/10 md:inline-block" dir="ltr">
   284→              {t("accountNo")} {c.accountNo}
   285→            </span>
   286→          </div>
   287→          <div className="flex shrink-0 items-center gap-1.5 sm:gap-2">
   288→            <button
   289→              onClick={onLangToggle}
   290→              className="flex h-9 items-center rounded-lg border border-white/10 px-2.5 text-[11.5px] font-bold text-white/60 transition-colors hover:bg-white/[0.05] hover:text-white sm:px-3 sm:text-[12px]"
   291→              aria-label="Toggle language"
   292→            >
   293→              {lang === "en" ? "العربية" : "EN"}
   294→            </button>
   295→            <button
   296→              onClick={() => go("notifications")}
   297→              aria-label={t("notificationsTitle")}
   298→              className="relative flex h-9 w-9 items-center justify-center rounded-lg border border-white/10 text-white/60 transition-colors hover:bg-white/[0.05] hover:text-white"
   299→            >
   300→              <Bell className="h-4 w-4" />
   301→              {unread > 0 && (
   302→                <span className="absolute -right-1 -top-1 flex h-4 min-w-4 items-center justify-center rounded-full bg-amber-400 px-1 text-[9.5px] font-bold text-[#04121c]">{unread}</span>
   303→              )}
   304→            </button>
   305→            <div className="hidden items-center gap-2.5 rounded-full border border-white/10 py-1 pe-3 ps-1 sm:flex">
   306→              <span className="flex h-7 w-7 items-center justify-center rounded-full bg-[#00E5A0] text-[11px] font-bold text-[#04121c]">{initials(c.name)}</span>
   307→              <span className="hidden text-left leading-tight md:block" dir="auto">
   308→                <span className="block max-w-[120px] truncate text-[12px] font-bold text-white">{c.name}</span>
   309→                <span className="block text-[10.5px] font-semibold text-white/40">{t("privateClient")}</span>
   310→              </span>
   311→            </div>
   312→            <button
   313→              onClick={signOut}
   314→              aria-label={t("signOut")}
   315→              className="flex h-9 items-center gap-1.5 rounded-lg border border-white/10 px-2.5 text-[12px] font-semibold text-white/60 transition-colors hover:bg-white/[0.05] hover:text-white sm:px-3"
   316→            >
   317→              <LogOut className="h-3.5 w-3.5" />
   318→              <span className="hidden md:inline">{t("signOut")}</span>
   319→            </button>
   320→          </div>
   321→        </div>
   322→      </header>
   323→
   324→      {/* ---------- body: sidebar + content ---------- */}
   325→      <div className="relative mx-auto flex w-full max-w-[1280px] items-start gap-6 px-0 sm:px-6">
   326→        {/* desktop sidebar */}
   327→        <aside className="sticky top-[72px] hidden w-[218px] shrink-0 py-6 lg:block" aria-label="Client menu">
   328→          <nav className="overflow-hidden rounded-2xl border border-white/[0.07] bg-[#071923]/85 p-2">
   329→            {NAV.map((item) => (
   330→              <button
   331→                key={item.id}
   332→                onClick={() => go(item.id)}
   333→                aria-current={section === item.id ? "page" : undefined}
   334→                className={cn(
   335→                  "flex w-full items-center gap-2.5 rounded-xl px-3 py-2.5 text-[13px] font-semibold transition-colors",
   336→                  section === item.id ? "bg-[#00E5A0]/12 text-[#00E5A0]" : "text-white/55 hover:bg-white/[0.04] hover:text-white",
   337→                )}
   338→              >
   339→                {item.icon}
   340→                <span className="flex-1 text-start">{item.label}</span>
   341→                {!!item.badge && item.badge > 0 && (
   342→                  <span className="flex h-4.5 min-w-[18px] items-center justify-center rounded-full bg-amber-400 px-1 text-[10px] font-bold text-[#04121c]">{item.badge}</span>
   343→                )}
   344→              </button>
   345→            ))}
   346→            <div className="my-2 h-px bg-white/[0.06]" />
   347→            <button
   348→              onClick={signOut}
   349→              className="flex w-full items-center gap-2.5 rounded-xl px-3 py-2.5 text-[13px] font-semibold text-white/55 transition-colors hover:bg-white/[0.04] hover:text-white"
   350→            >
   351→              <LogOut className="h-4 w-4" />
   352→              <span className="flex-1 text-start">{t("signOut")}</span>
   353→            </button>
   354→          </nav>
   355→          <p className="mt-3 px-3 text-[10.5px] leading-relaxed text-white/25" dir="ltr">
   356→            {t("accountNo")} {c.accountNo}
   357→          </p>
   358→        </aside>
   359→
   360→        {/* content */}
   361→        <main className="min-w-0 flex-1 py-5 sm:py-6">
   362→          {section === "overview" && (
   363→            <OverviewSection
   364→              view={view}
   365→              t={t}
   366→              lang={lang}
   367→              coins={coins}
   368→              stocks={stocks}
   369→              holdingRows={holdingRows}
   370→              invested={invested}
   371→              cash={cash}
   372→              total={total}
   373→              available={available}
   374→              portfolioChange={portfolioChange}
   375→              totalDeposits={totalDeposits}
   376→              pendingWithdrawalTotal={pendingWithdrawalTotal}
   377→              history={history}
   378→              allocation={allocation}
   379→              suspended={!!suspended}
   380→              onDeposit={() => !suspended && setReqModal("deposit")}
   381→              onWithdraw={() => !suspended && setReqModal("withdrawal")}
   382→              onGo={go}
   383→            />
   384→          )}
   385→
   386→          {section === "portfolio" && (
   387→            <div className="grid grid-cols-1 gap-5">
   388→              <SectionHeading title={t("menuPortfolio")} sub={t("welcomeSub")} />
   389→              <HoldingsTable rows={holdingRows} t={t} lang={lang} />
   390→              <AllocationBar segments={allocation} t={t} />
   391→              <MarketStrip coins={coins} stocks={stocks} t={t} />
   392→            </div>
   393→          )}
   394→
   395→          {section === "transactions" && (
   396→            <div className="grid grid-cols-1 gap-5">
   397→              <SectionHeading title={t("txHistory")} sub={`${txs.length} · ${t("allTime")}`} />
   398→              <div className="flex flex-wrap items-center justify-between gap-2.5">
   399→                <div className="flex items-center gap-1 rounded-lg bg-white/[0.05] p-1">
   400→                  {(
   401→                    [
   402→                      ["all", t("filterAll")],
   403→                      ["in", t("filterIn")],
   404→                      ["out", t("filterOut")],
   405→                    ] as const
   406→                  ).map(([key, label]) => (
   407→                    <button
   408→                      key={key}
   409→                      onClick={() => setTxFilter(key)}
   410→                      className={cn("rounded-md px-2.5 py-1.5 text-[11.5px] font-bold transition-colors", txFilter === key ? "bg-[#00E5A0]/15 text-[#00E5A0]" : "text-white/50 hover:text-white")}
   411→                    >
   412→                      {label}
   413→                    </button>
   414→                  ))}
   415→                </div>
   416→                <button onClick={downloadCsv} className="inline-flex items-center gap-1.5 rounded-lg border border-white/10 px-2.5 py-2 text-[11.5px] font-bold text-white/60 transition-colors hover:bg-white/[0.05] hover:text-white">
   417→                  <Download className="h-3.5 w-3.5" /> CSV
   418→                </button>
   419→              </div>
   420→              <div className="overflow-hidden rounded-2xl border border-white/[0.07] bg-[#071923]/85">
   421→                <ul className="divide-y divide-white/[0.05]">
   422→                  {(showAllTxs ? filteredTxs : filteredTxs.slice(0, 12)).map((tx) => (
   423→                    <TxRowItem key={tx.id} tx={tx} t={t} lang={lang} />
   424→                  ))}
   425→                  {filteredTxs.length === 0 && (
   426→                    <li className="px-6 py-12 text-center text-[13px] text-white/45">{t("noTx")}</li>
   427→                  )}
   428→                </ul>
   429→                {filteredTxs.length > 12 && (
   430→                  <button onClick={() => setShowAllTxs((v) => !v)} className="w-full border-t border-white/[0.06] py-3 text-[12.5px] font-bold text-[#00E5A0] transition-colors hover:bg-[#00E5A0]/[0.06]">
   431→                    {showAllTxs ? "−" : "+"} {t("viewAllTx")} ({filteredTxs.length})
   432→                  </button>
   433→                )}
   434→              </div>
   435→            </div>
   436→          )}
   437→
   438→          {section === "deposits" && (
   439→            <DepositsSection t={t} lang={lang} deposits={deposits} totalDeposits={totalDeposits} suspended={!!suspended} onDeposit={() => !suspended && setReqModal("deposit")} />
   440→          )}
