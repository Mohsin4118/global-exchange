    40→    40→export function fmtDate(iso: string, lang: Lang = "en"): string {
    41→    41→  const d = new Date(`${iso}T00:00:00`);
    42→    42→  if (Number.isNaN(d.getTime())) return iso;
    43→    43→  try {
    44→    44→    return d.toLocaleDateString(lang === "ar" ? "ar" : "en-US", { month: "short", day: "numeric", year: "numeric" });
    45→    45→  } catch {
    46→    46→    return iso;

    50→    50→export function seedFor(id: string): number {
    51→    51→  let s = 0;
    52→    52→  for (let i = 0; i < id.length; i++) s = (s * 31 + id.charCodeAt(i)) % 233280;
    53→    53→  return s || 7;
    54→    54→}

   130→   130→  const max = Math.max(...pts);
   131→   131→  const min = Math.min(...pts);
   132→   132→  const range = max - min || Math.max(1, max * 0.002);
   133→   133→  const W = 560;
   134→   134→  const H = 190;
   135→   135→  const step = W / (pts.length - 1);
   136→   136→  const coords = pts.map((p, i) => [i * step, H - 26 - ((p - min) / range) * (H - 52)] as const);
   137→   137→  let path = `M${coords[0][0]},${coords[0][1]}`;
   138→   138→  for (let i = 1; i < coords.length; i++) {
   139→   139→    const [x0, y0] = coords[i - 1];
   140→   140→    const [x1, y1] = coords[i];

   142→   142→    path += ` C${cx},${y0} ${cx},${y1} ${x1},${y1}`;

   189→   189→export interface HoldingRow {
   190→   190→  units: number;

   195→   195→
   196→   196→export function buildHoldingRows(holdings: ClientView["client"]["holdings"], markets: Coin[]): HoldingRow[] {
   197→   197→  return holdings
   198→   198→    .map((h) => {
   199→   199→      const coin = markets.find((m) => m.id === h.assetId);
   200→   200→      if (!coin) return null;
   201→   201→      const base = sparkline(seedFor(coin.id), 24, 1.15);
   202→   202→      const trend = coin.change24h >= 0 ? 7 : -7;
   203→   203→      const series = base.map((v, i) => v + (i / (base.length - 1)) * trend);
   204→   204→      return { units: h.units, coin, value: coin.price * h.units, sparkPath: sparklinePath(series, 72, 28) };
   205→   205→    })
   206→   206→    .filter((r): r is HoldingRow => r !== null)
   207→   207→    .sort((a, b) => b.value - a.value);
   208→   208→}

   341→   341→export function RequestModal({
   342→   342→  kind,
   343→   343→  t,
   344→   344→  available,
   345→   345→  onClose,
   346→   346→  onSubmit,
   347→   347→}: {
   348→   348→  kind: "deposit" | "withdrawal";
   349→   349→  t: T;
   350→   350→  available: number;
   351→   351→  onClose: () => void;
   352→   352→  onSubmit: (amount: number, note: string) => Promise<string | null>; // returns error key or null
   353→   353→}) {
   354→   354→  const [amount, setAmount] = useState("");
   355→   355→  const [note, setNote] = useState("");
   356→   356→  const [error, setError] = useState<string | null>(null);
   357→   357→  const [busy, setBusy] = useState(false);
   358→   358→
   359→   359→  const submit = async () => {
   360→   360→    const value = parseFloat(amount);
   361→   361→    if (!Number.isFinite(value) || value <= 0) {
   362→   362→      setError(t("requestInvalid"));
   363→   363→      return;
   364→   364→    }
   365→   365→    if (kind === "withdrawal" && value > available) {
   366→   366→      setError(t("requestTooMuch"));
   367→   367→      return;
   368→   368→    }
   369→   369→    setBusy(true);
   370→   370→    const err = await onSubmit(value, note.trim());
   371→   371→    setBusy(false);
   372→   372→    if (err) setError(t(err as StringKey));
   373→   373→  };
   374→   374→
   375→   375→  return (
   376→   376→    <div className="fixed inset-0 z-50 flex items-center justify-center p-4">
   377→   377→      <div className="absolute inset-0 bg-slate-950/50 backdrop-blur-[2px]" onClick={onClose} />
   378→   378→      <div className="relative z-10 w-full max-w-md overflow-hidden rounded-2xl bg-white shadow-2xl">
   379→   379→        <div className="flex items-center justify-between p-5 pb-0">
   380→   380→          <h3 className="text-lg font-bold text-slate-900">{kind === "deposit" ? t("requestTitleDeposit") : t("requestTitleWithdraw")}</h3>
   381→   381→          <button onClick={onClose} aria-label="Close" className="flex h-8 w-8 items-center justify-center rounded-lg text-slate-400 transition-colors hover:bg-slate-100 hover:text-slate-700">
   382→   382→            <X className="h-5 w-5" />
   383→   383→          </button>
   384→   384→        </div>
   385→   385→        <div className="space-y-4 p-5">
   386→   386→          <p className="rounded-lg bg-slate-50 px-3.5 py-2.5 text-[12.5px] leading-relaxed text-slate-500">{t("requestHint")}</p>
   387→   387→          <label className="block">
