    36→export function hashPassword(password: string): { salt: string; hash: string } {
    37→  const salt = crypto.randomBytes(16).toString("hex");
    38→  const hash = crypto.scryptSync(password, salt, 64).toString("hex");
    39→  return { salt, hash };
    40→}
    41→
    42→export function verifyPassword(password: string, stored: { salt: string; hash: string }): boolean {
    43→  try {
    44→    const candidate = crypto.scryptSync(password, stored.salt, 64);
    45→    const expected = Buffer.from(stored.hash, "hex");
    46→    return candidate.length === expected.length && crypto.timingSafeEqual(candidate, expected);
    47→  } catch {
    48→    return false;
    49→  }
    50→}

    55→export function uid(prefix: string): string {
    56→  seq += 1;
    57→  return `${prefix}${Date.now().toString(36)}${seq.toString(36)}${crypto.randomBytes(3).toString("hex")}`;
    58→}

    60→export function newReference(): string {
    61→  const d = new Date();
    62→  const ymd = `${d.getFullYear().toString().slice(-2)}${String(d.getMonth() + 1).padStart(2, "0")}${String(d.getDate()).padStart(2, "0")}`;
    63→  return `TXN-${ymd}-${crypto.randomInt(1000, 9999)}`;
    64→}

    66→export function stamp(): string {
    67→  const d = new Date();
    68→  const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
    69→  let h = d.getHours();
    70→  const ampm = h >= 12 ? "PM" : "AM";
    71→  h = h % 12 || 12;
    72→  return `${months[d.getMonth()]} ${d.getDate()}, ${d.getFullYear()}, ${String(h).padStart(2, "0")}:${String(d.getMinutes()).padStart(2, "0")} ${ampm}`;
    73→}

    75→function relTime(iso: string): string {
    76→  const then = new Date(iso).getTime();
    77→  if (Number.isNaN(then)) return "just now";
    78→  const mins = Math.max(0, Math.round((Date.now() - then) / 60000));
    79→  if (mins < 1) return "just now";
    80→  if (mins < 60) return `${mins}m ago`;
    81→  const hours = Math.round(mins / 60);
    82→  if (hours < 24) return `${hours}h ago`;
    83→  const days = Math.round(hours / 24);
    84→  if (days < 7) return `${days}d ago`;
    85→  return `${Math.round(days / 7)}w ago`;
    86→}
    87→
    88→function daysAgoISO(days: number): string {
    89→  const d = new Date();
    90→  d.setDate(d.getDate() - days);
    91→  return d.toISOString().slice(0, 10);
    92→}

    94→function hashCode(s: string): number {
    95→  let h = 0;
    96→  for (let i = 0; i < s.length; i++) h = (h * 31 + s.charCodeAt(i)) | 0;
    97→  return h;
    98→}

   102→function makeTx(p: {
   103→  clientId: string;
   104→  dateISO: string;
   105→  kind: Tx["kind"];
   106→  type: Tx["type"];
   107→  amount: number;
   108→  label: string;
   109→  labelKey?: string;
   110→  asset?: string;
   111→  status?: Tx["status"];
   112→  method?: string;
   113→  notes?: string;
   114→}): Tx {
   115→  return {
   116→    id: uid("tx"),
   117→    clientId: p.clientId,
   118→    dateISO: p.dateISO,
   119→    kind: p.kind,
   120→    type: p.type,
   121→    amount: p.amount,
   122→    label: p.label,
   123→    labelKey: p.labelKey,
   124→    asset: p.asset,
   125→    status: p.status ?? "COMPLETED",
   126→    reference: newReference(),
   127→    method: p.method ?? "Bank Transfer",
   128→    notes: p.notes ?? "—",
   129→    createdAtISO: new Date(`${p.dateISO}T10:00:00Z`).toISOString(),
   130→    updatedAtISO: new Date(`${p.dateISO}T10:00:00Z`).toISOString(),
   131→  };
   132→}

   134→function makeNotification(p: Pick<Notification, "audience" | "title" | "body" | "kind"> & { createdAtISO?: string; unread?: boolean }): Notification {
   135→  const createdAtISO = p.createdAtISO ?? new Date().toISOString();

   141→    kind: p.kind,
   142→    createdAtISO,
   143→    time: relTime(createdAtISO),
   144→    unread: p.unread ?? true,
   145→  };
   146→}

   417→function readDb(): DbData {
   418→  if (!fs.existsSync(DB_FILE)) {
   419→    fs.mkdirSync(DATA_DIR, { recursive: true });
   420→    const seeded = seed();
   421→    fs.writeFileSync(DB_FILE, JSON.stringify(seeded, null, 2), "utf8");
   422→    return seeded;
   423→  }
   424→  try {
   425→    const raw = fs.readFileSync(DB_FILE, "utf8");
   426→    const parsed = JSON.parse(raw) as DbData;
   427→    if (!parsed.adminUser || !Array.isArray(parsed.clients)) throw new Error("corrupt db");
   428→    parsed.sessions ??= [];
   429→    parsed.comments ??= [];
   430→    parsed.staff ??= [];
   431→    parsed.roles ??= [];
   432→    parsed.audit ??= [];
   433→    parsed.notifications ??= [];
   434→    parsed.txs ??= [];
   435→    return parsed;
   436→  } catch {
   437→    const seeded = seed();
   438→    fs.writeFileSync(DB_FILE, JSON.stringify(seeded, null, 2), "utf8");
   439→    return seeded;
   440→  }
   441→}
   442→
   443→function writeDb(data: DbData) {
   444→  fs.mkdirSync(DATA_DIR, { recursive: true });
   445→  const tmp = `${DB_FILE}.tmp`;
   446→  fs.writeFileSync(tmp, JSON.stringify(data, null, 2), "utf8");
   447→  fs.renameSync(tmp, DB_FILE);
   448→}

   450→/** Read (seeding on first use). */
   451→export function db(): DbData {
   452→  return readDb();
   453→}

   455→/** Mutate the database atomically and persist. */
   456→export function mutate<T>(fn: (data: DbData) => T): T {
   457→  const data = readDb();
   458→  const result = fn(data);
   459→  writeDb(data);
   460→  return result;
   461→}

   506→export function round2(n: number): number {
   507→  return Math.round((n + Number.EPSILON) * 100) / 100;
   508→}

   510→export function computeFinancials(clientId: string, data: DbData): ComputedFinancials {
   511→  const client = data.clients.find((c) => c.id === clientId);
   512→  const own = data.txs.filter((t) => t.clientId === clientId);
   513→  let credits = 0;
   514→  let debits = 0;
   515→  let pendingWithdrawals = 0;
   516→  for (const t of own) {
   517→    if (t.status === "COMPLETED") {
   518→      if (t.type === "CREDIT") credits += t.amount;
   519→      else debits += t.amount;
   520→    } else if ((t.status === "PENDING" || t.status === "PROCESSING") && t.kind === "withdrawal" && t.type === "DEBIT") {
   521→      pendingWithdrawals += t.amount;
   522→    }
   523→  }
   524→  const balance = Math.max(0, (client?.openingBalance ?? 0) + credits - debits);
   525→  return {
   526→    balance: round2(balance),
   527→    available: round2(Math.max(0, balance - pendingWithdrawals)),
   528→    credits: round2(credits),
   529→    debits: round2(debits),
   530→    pendingWithdrawals: round2(pendingWithdrawals),
   531→    txCount: own.length,
   532→  };
   533→}

   535→/** Balance timeline per client so each tx row can show "balance after". */
   536→function balanceAfterMap(data: DbData): Map<string, number> {
   537→  const result = new Map<string, number>();
   538→  const byClient = new Map<string, Tx[]>();
   539→  for (const t of data.txs) {
   540→    const list = byClient.get(t.clientId) ?? [];
   541→    list.push(t);
   542→    byClient.set(t.clientId, list);
   543→  }
   544→  for (const [clientId, list] of byClient) {
   545→    const client = data.clients.find((c) => c.id === clientId);
   546→    const sorted = [...list].sort((a, b) => (a.dateISO === b.dateISO ? a.createdAtISO.localeCompare(b.createdAtISO) : a.dateISO.localeCompare(b.dateISO)));
   547→    let bal = client?.openingBalance ?? 0;
   548→    for (const t of sorted) {
   549→      if (t.status === "COMPLETED") bal = Math.max(0, bal + (t.type === "CREDIT" ? t.amount : -t.amount));
   550→      result.set(t.id, round2(bal));
   551→    }
   552→  }
   553→  return result;
   554→}

   563→export function adminSnapshot(): AdminSnapshot {
   564→  const data = readDb();
   565→  const balAfter = balanceAfterMap(data);
   566→  const clients = data.clients.map((c) => ({
   567→    ...publicClient(c),
   568→    financials: computeFinancials(c.id, data),
   569→    hasPortalPassword: true,
   570→  }));
   571→  const nameById = new Map(data.clients.map((c) => [c.id, c.name] as const));
   572→  const txs = [...data.txs]
   573→    .sort((a, b) => (a.dateISO === b.dateISO ? b.createdAtISO.localeCompare(a.createdAtISO) : b.dateISO.localeCompare(a.dateISO)))
   574→    .map((t) => ({ ...t, clientName: nameById.get(t.clientId) ?? "—", balanceAfter: balAfter.get(t.id) ?? 0 }));
   575→
   576→  const completed = txs.filter((t) => t.status === "COMPLETED");
   577→  const stats: AdminStats = {
   578→    totalClients: clients.length,
   579→    activeClients: clients.filter((c) => c.status === "active").length,
   580→    totalBalance: round2(clients.reduce((s, c) => s + c.financials.balance, 0)),
   581→    volume: round2(completed.reduce((s, t) => s + t.amount, 0)),
   582→    credits: round2(completed.filter((t) => t.type === "CREDIT").reduce((s, t) => s + t.amount, 0)),
   583→    debits: round2(completed.filter((t) => t.type === "DEBIT").reduce((s, t) => s + t.amount, 0)),
   584→    pendingWithdrawals: txs.filter((t) => t.kind === "withdrawal" && (t.status === "PENDING" || t.status === "PROCESSING")).length,
   585→    txCount: txs.length,
   586→  };
   587→
   588→  const comments: Record<string, ClientComment[]> = {};
   589→  for (const cm of data.comments) {
   590→    (comments[cm.clientId] ??= []).push(cm);
   591→  }
   592→
   593→  return {
   594→    clients,
   595→    txs,
   596→    notifications: [...data.notifications].sort((a, b) => b.createdAtISO.localeCompare(a.createdAtISO)),
   597→    audit: data.audit,
   598→    comments,
   599→    staff: data.staff,
   600→    roles: data.roles,
   601→    stats,
   602→  };
   603→}

   606→export function clientView(clientId: string): { ok: true; view: { client: PublicClient; financials: ComputedFinancials; txs: Tx[]; notifications: Notification[] } } | { ok: false } {
   607→  const data = readDb();

   609→  if (!client) return { ok: false };
   610→  return {
   611→    ok: true,
   612→    view: {
   613→      client: publicClient(client),
   614→      financials: computeFinancials(clientId, data),
   615→      txs: data.txs
   616→        .filter((t) => t.clientId === clientId)
   617→        .sort((a, b) => (a.dateISO === b.dateISO ? b.createdAtISO.localeCompare(a.createdAtISO) : b.dateISO.localeCompare(a.dateISO))),
   618→      notifications: data.notifications.filter((n) => n.audience === clientId).sort((a, b) => b.createdAtISO.localeCompare(a.createdAtISO)),
   619→    },
   620→  };
   621→}
