    38→export function hashPassword(password: string): { salt: string; hash: string } {
    39→  const salt = crypto.randomBytes(16).toString("hex");
    40→  const hash = crypto.scryptSync(password, salt, 64).toString("hex");
    41→  return { salt, hash };
    42→}
    43→
    44→export function verifyPassword(password: string, stored: { salt: string; hash: string }): boolean {
    45→  try {
    46→    const candidate = crypto.scryptSync(password, stored.salt, 64);
    47→    const expected = Buffer.from(stored.hash, "hex");
    48→    return candidate.length === expected.length && crypto.timingSafeEqual(candidate, expected);
    49→  } catch {
    50→    return false;
    51→  }
    52→}

    56→let seq = 0;
    57→export function uid(prefix: string): string {
    58→  seq += 1;
    59→  return `${prefix}${Date.now().toString(36)}${seq.toString(36)}${crypto.randomBytes(3).toString("hex")}`;
    60→}

    62→export function newReference(): string {
    63→  const d = new Date();
    64→  const ymd = `${d.getFullYear().toString().slice(-2)}${String(d.getMonth() + 1).padStart(2, "0")}${String(d.getDate()).padStart(2, "0")}`;
    65→  return `TXN-${ymd}-${crypto.randomInt(1000, 9999)}`;
    66→}

    68→export function stamp(): string {
    69→  const d = new Date();
    70→  const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
    71→  let h = d.getHours();
    72→  const ampm = h >= 12 ? "PM" : "AM";
    73→  h = h % 12 || 12;
    74→  return `${months[d.getMonth()]} ${d.getDate()}, ${d.getFullYear()}, ${String(h).padStart(2, "0")}:${String(d.getMinutes()).padStart(2, "0")} ${ampm}`;
    75→}

    77→function relTime(iso: string): string {
    78→  const then = new Date(iso).getTime();
    79→  if (Number.isNaN(then)) return "just now";
    80→  const mins = Math.max(0, Math.round((Date.now() - then) / 60000));
    81→  if (mins < 1) return "just now";
    82→  if (mins < 60) return `${mins}m ago`;
    83→  const hours = Math.round(mins / 60);
    84→  if (hours < 24) return `${hours}h ago`;
    85→  const days = Math.round(hours / 24);
    86→  if (days < 7) return `${days}d ago`;
    87→  return `${Math.round(days / 7)}w ago`;
    88→}
    89→
    90→function daysAgoISO(days: number): string {
    91→  const d = new Date();
    92→  d.setDate(d.getDate() - days);
    93→  return d.toISOString().slice(0, 10);
    94→}

    96→function hashCode(s: string): number {
    97→  let h = 0;
    98→  for (let i = 0; i < s.length; i++) h = (h * 31 + s.charCodeAt(i)) | 0;
    99→  return h;
   100→}

   104→function makeTx(p: {
   105→  clientId: string;
   106→  dateISO: string;
   107→  kind: Tx["kind"];
   108→  type: Tx["type"];
   109→  amount: number;
   110→  label: string;
   111→  labelKey?: string;
   112→  asset?: string;
   113→  status?: Tx["status"];
   114→  method?: string;
   115→  notes?: string;
   116→}): Tx {
   117→  const status = p.status ?? "COMPLETED";
   118→  return {
   119→    id: uid("tx"),
   120→    clientId: p.clientId,
   121→    dateISO: p.dateISO,
   122→    kind: p.kind,
   123→    type: p.type,
   124→    amount: p.amount,
   125→    label: p.label,
   126→    labelKey: p.labelKey,
   127→    asset: p.asset,
   128→    status,
   129→    reference: newReference(),
   130→    method: p.method ?? "Bank Transfer",
   131→    notes: p.notes ?? "—",
   132→    history: [{ at: new Date(`${p.dateISO}T10:00:00Z`).toISOString(), by: "CryptoWise", byRole: "system", from: null, to: status }],
   133→    createdAtISO: new Date(`${p.dateISO}T10:00:00Z`).toISOString(),
   134→    updatedAtISO: new Date(`${p.dateISO}T10:00:00Z`).toISOString(),
   135→  };
   136→}

   138→function makeNotification(p: Pick<Notification, "audience" | "title" | "body" | "kind"> & { createdAtISO?: string; unread?: boolean }): Notification {
   139→  const createdAtISO = p.createdAtISO ?? new Date().toISOString();

   145→    kind: p.kind,
   146→    createdAtISO,
   147→    time: relTime(createdAtISO),
   148→    unread: p.unread ?? true,
   149→  };
   150→}

   421→function readDb(): DbData {
   422→  if (!fs.existsSync(DB_FILE)) {
   423→    fs.mkdirSync(DATA_DIR, { recursive: true });
   424→    const seeded = seed();
   425→    fs.writeFileSync(DB_FILE, JSON.stringify(seeded, null, 2), "utf8");
   426→    return seeded;
   427→  }
   428→  try {
   429→    const raw = fs.readFileSync(DB_FILE, "utf8");
   430→    const parsed = JSON.parse(raw) as DbData;
   431→    if (!parsed.adminUser || !Array.isArray(parsed.clients)) throw new Error("corrupt db");
   432→    parsed.sessions ??= [];
   433→    parsed.comments ??= [];
   434→    parsed.staff ??= [];
   435→    parsed.roles ??= [];
   436→    parsed.audit ??= [];
   437→    parsed.notifications ??= [];
   438→    parsed.txs ??= [];
   439→    return parsed;
   440→  } catch {
   441→    const seeded = seed();
   442→    fs.writeFileSync(DB_FILE, JSON.stringify(seeded, null, 2), "utf8");
   443→    return seeded;
   444→  }
   445→}

   447→function writeDb(data: DbData) {
   448→  fs.mkdirSync(DATA_DIR, { recursive: true });
   449→  const tmp = `${DB_FILE}.tmp`;
   450→  fs.writeFileSync(tmp, JSON.stringify(data, null, 2), "utf8");
   451→  fs.renameSync(tmp, DB_FILE);
   452→}

   454→/** Read (seeding on first use). */
   455→export function db(): DbData {
   456→  return readDb();
   457→}

   460→export function mutate<T>(fn: (data: DbData) => T): T {
   461→  const data = readDb();
   462→  const result = fn(data);
   463→  writeDb(data);
   464→  return result;

   514→/** Request statuses that are still moving through the pipeline. Everything
   515→    except REJECTED / CANCELLED / COMPLETED reserves funds (withdrawals) and
   516→    counts as an open request. */
   517→export function inFlight(status: TxStatus): boolean {
   518→  return status === "PENDING" || status === "UNDER_REVIEW" || status === "APPROVED" || status === "PROCESSING";

   521→export function computeFinancials(clientId: string, data: DbData): ComputedFinancials {
   522→  const client = data.clients.find((c) => c.id === clientId);
   523→  const own = data.txs.filter((t) => t.clientId === clientId);
   524→  let credits = 0;
   525→  let debits = 0;
   526→  let pendingWithdrawals = 0;
   527→  for (const t of own) {
   528→    if (t.status === "COMPLETED") {
   529→      if (t.type === "CREDIT") credits += t.amount;
   530→      else debits += t.amount;
   531→    } else if (inFlight(t.status) && t.kind === "withdrawal" && t.type === "DEBIT") {
   532→      pendingWithdrawals += t.amount;
   533→    }
   534→  }
   535→  const balance = Math.max(0, (client?.openingBalance ?? 0) + credits - debits);
   536→  return {
   537→    balance: round2(balance),
   538→    available: round2(Math.max(0, balance - pendingWithdrawals)),
   539→    credits: round2(credits),
   540→    debits: round2(debits),
   541→    pendingWithdrawals: round2(pendingWithdrawals),
   542→    txCount: own.length,
   543→  };
   544→}

   546→/** Balance timeline per client so each tx row can show "balance after". */
   547→function balanceAfterMap(data: DbData): Map<string, number> {
   548→  const result = new Map<string, number>();
   549→  const byClient = new Map<string, Tx[]>();
   550→  for (const t of data.txs) {
   551→    const list = byClient.get(t.clientId) ?? [];
   552→    list.push(t);
   553→    byClient.set(t.clientId, list);
   554→  }
   555→  for (const [clientId, list] of byClient) {
   556→    const client = data.clients.find((c) => c.id === clientId);
   557→    const sorted = [...list].sort((a, b) => (a.dateISO === b.dateISO ? a.createdAtISO.localeCompare(b.createdAtISO) : a.dateISO.localeCompare(b.dateISO)));
   558→    let bal = client?.openingBalance ?? 0;
   559→    for (const t of sorted) {
   560→      if (t.status === "COMPLETED") bal = Math.max(0, bal + (t.type === "CREDIT" ? t.amount : -t.amount));
   561→      result.set(t.id, round2(bal));
   562→    }
   563→  }
   564→  return result;
   565→}

   574→export function adminSnapshot(): AdminSnapshot {
   575→  const data = readDb();
   576→  const balAfter = balanceAfterMap(data);
   577→  const clients = data.clients.map((c) => ({
   578→    ...publicClient(c),
   579→    financials: computeFinancials(c.id, data),
   580→    hasPortalPassword: true,
   581→  }));
   582→  const nameById = new Map(data.clients.map((c) => [c.id, c.name] as const));
   583→  const txs = [...data.txs]
   584→    .sort((a, b) => (a.dateISO === b.dateISO ? b.createdAtISO.localeCompare(a.createdAtISO) : b.dateISO.localeCompare(a.dateISO)))
   585→    .map((t) => ({ ...t, clientName: nameById.get(t.clientId) ?? "—", balanceAfter: balAfter.get(t.id) ?? 0 }));
   586→
   587→  const completed = txs.filter((t) => t.status === "COMPLETED");
   588→  const requestCounts = txs.reduce(
   589→    (acc, t) => {
   590→      acc[t.status] = (acc[t.status] ?? 0) + 1;
   591→      return acc;
   592→    },
   593→    { COMPLETED: 0, PENDING: 0, UNDER_REVIEW: 0, APPROVED: 0, PROCESSING: 0, REJECTED: 0, CANCELLED: 0 } as Record<TxStatus, number>,
   594→  );
   595→  const stats: AdminStats = {
   596→    totalClients: clients.length,
   597→    activeClients: clients.filter((c) => c.status === "active").length,
   598→    totalBalance: round2(clients.reduce((s, c) => s + c.financials.balance, 0)),
   599→    volume: round2(completed.reduce((s, t) => s + t.amount, 0)),
   600→    credits: round2(completed.filter((t) => t.type === "CREDIT").reduce((s, t) => s + t.amount, 0)),
   601→    debits: round2(completed.filter((t) => t.type === "DEBIT").reduce((s, t) => s + t.amount, 0)),
   602→    pendingWithdrawals: txs.filter((t) => t.kind === "withdrawal" && inFlight(t.status)).length,
   603→    txCount: txs.length,
   604→    requestCounts,
   605→  };
   606→
   607→  const comments: Record<string, ClientComment[]> = {};
   608→  for (const cm of data.comments) {
   609→    (comments[cm.clientId] ??= []).push(cm);
   610→  }
   611→
   612→  return {
   613→    clients,
   614→    txs,
   615→    notifications: [...data.notifications].sort((a, b) => b.createdAtISO.localeCompare(a.createdAtISO)),
   616→    audit: data.audit,
   617→    comments,
   618→    staff: data.staff,
   619→    roles: data.roles,
   620→    stats,
   621→  };
   622→}

   625→export function clientView(clientId: string): { ok: true; view: { client: PublicClient; financials: ComputedFinancials; txs: Tx[]; notifications: Notification[] } } | { ok: false } {
   626→  const data = readDb();

   637→        .map(clientSafeTx),
   638→      notifications: data.notifications.filter((n) => n.audience === clientId).sort((a, b) => b.createdAtISO.localeCompare(a.createdAtISO)),
   639→    },
   640→  };
   641→}

   643→/** Client-safe transaction: strips the PRIVATE admin note and every internal
   644→    remark from the history (the status transition itself stays visible; only
   645→    the internal wording is removed). Admin actors are anonymised to
   646→    "Super Admin" so admin emails/details never leak through the client API (#30 / #36). */
   647→export function clientSafeTx(t: Tx): Tx {
   648→  const { adminNote: _private, ...rest } = t;
   649→  const history = (t.history ?? []).map<TxEvent>((e) => ({
   650→    at: e.at,
   651→    by: e.byRole === "admin" ? "Super Admin" : e.by,
   652→    byRole: e.byRole,
   653→    from: e.from,
   654→    to: e.to,
   655→    ...(e.note ? { note: e.note } : {}),
   656→    ...(e.changes ? { changes: e.changes } : {}),
   657→  }));
   658→  return { ...rest, history };
   659→}
